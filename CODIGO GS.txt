function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000); 
    
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var userId = data.userId;
    var payload = data.datos || {};

    if (!userId) return ContentService.createTextOutput(JSON.stringify({ error: "Falta userId" })).setMimeType(ContentService.MimeType.JSON);

    var resultado = "";
    switch (action) {
      // MÉTODOS BÁSICOS DE USUARIO
      case "getDashboardData": resultado = getDashboardData(userId, payload.email); break;
      case "saveWelcomeData": resultado = saveWelcomeData(payload, userId); break;
      case "guardarEscalon": resultado = guardarEscalon(payload, userId); break;
      case "guardarFotoEscalonBackend": resultado = guardarFotoEscalonBackend(payload, userId); break;
      case "saveMedidasMesUniversal": resultado = saveMedidasMesUniversal(payload, userId); break;
      case "saveDiaDiario": resultado = saveDiaDiario(payload, userId); break;
      case "registrarTelefono": resultado = registrarTelefono(payload, userId); break;

      case "actualizarIconoApp": 
        if (userId !== ADMIN_UID) throw "No autorizado";
        resultado = actualizarIconoApp(payload.url); break;
        
      // ADMIN Y SUPER ADMIN (Usuarios y Finanzas Original)
      case "getUsersList": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = getUsersList(); break;
      case "cambiarEstadoUser": 
        if (!isUserAdmin(userId)) throw "No autorizado";

      case "eliminarUser":

      case "getFinanzasData": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = getFinanzasData(); break;
      case "registrarPago": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = registrarPago(payload); break;
      case "eliminarPago": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = eliminarPago(payload); break;
      case "editarPagoBackend": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = editarPagoBackend(payload); break;
        
      // NUEVOS CASOS SAAS, REPORTES Y SCRIPTS
      case "obtenerDatosControlSaaS": 
        if (!isUserAdmin(userId)) throw "No autorizado"; 
        resultado = obtenerDatosControlSaaS(); break;
      case "guardarCambioControlSaaS": 
        if (!isUserAdmin(userId)) throw "No autorizado"; 
        resultado = guardarCambioControlSaaS(payload.index, payload.tipo, payload.valor); break;
      case "generarSiguienteMesSaaS": 
        if (!isUserAdmin(userId)) throw "No autorizado"; 
        resultado = generarSiguienteMesSaaS(payload.index, payload.nuevoMonto); break;
      case "eliminarRegistroSaaS":

}

function registrarTelefono(payload, userId) {
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  dbSheet.getRange("R1").setValue("TELEFONO"); // Asegurar el título en R1
  var uids = dbSheet.getRange("A2:A").getValues();
  var targetRow = -1;
  for (var i = 0; i < uids.length; i++) { 
    if (uids[i][0] == userId) { targetRow = i + 2; break; } 
  }
  
  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    var estadoInicial = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    dbSheet.getRange(targetRow, 15).setValue(estadoInicial); 
    if (payload.email) dbSheet.getRange(targetRow, 16).setValue(payload.email); 
  }
  
  dbSheet.getRange(targetRow, 18).setValue(payload.telefono); // Columna R
  return "Teléfono guardado";
}

function getUserSpreadsheet(userId, email, nombre) {
  var centralSs = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = centralSs.getSheetByName("DB_Perfiles");
  var data = dbSheet.getRange("A2:Q").getValues(); 
  var targetRow = -1;
  var sheetId = "";
  var estado = "Autorizado";

  for (var i = 0; i < data.length; i++) {
    if (data[i][0] == userId) {
      targetRow = i + 2;
      estado = data[i][14] || "Autorizado"; 
      sheetId = data[i][16]; // Col Q
      
      if (email && data[i][15] === "") dbSheet.getRange(targetRow, 16).setValue(email);
      break;
    }
  }

  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    estado = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    dbSheet.getRange(targetRow, 15).setValue(estado); 
    if (email) dbSheet.getRange(targetRow, 16).setValue(email); 
  }

  if (!sheetId || sheetId === "") {
    if (userId === ADMIN_UID) {
      sheetId = TEMPLATE_ID;
      dbSheet.getRange(targetRow, 17).setValue(sheetId);
    } else {
      var template = DriveApp.getFileById(TEMPLATE_ID);
      var targetFolder = DriveApp.getFolderById(FOLDER_ID);
      var copy = template.makeCopy("Fitness Pro - " + (email || userId), targetFolder);
      sheetId = copy.getId();
      dbSheet.getRange(targetRow, 17).setValue(sheetId);
    }
  }

  return { ss: SpreadsheetApp.openById(sheetId), estado: estado, esNuevo: (sheetId === "") };
}

function parseFloatSafe(val) { if (val === "" || val === undefined || val === null) return ""; var str = String(val).replace(',', '.'); var num = parseFloat(str); return isNaN(num) ? "" : num; }
function readImageUrl(sheet, cellStr) { var formula = sheet.getRange(cellStr).getFormula(); if (formula && formula.toUpperCase().indexOf('=IMAGE') !== -1) { var match = formula.match(/=IMAGE\("([^"]+)"/i); if (match) return match[1]; } return sheet.getRange(cellStr).getDisplayValue(); }
function writeImageUrl(sheet, cellStr, url) { var range = sheet.getRange(cellStr); if (url && url.toString().startsWith("http")) { var directUrl = url; var regex = /drive\.google\.com\/(?:file\/d\/|open\?id=)([-_a-zA-Z0-9]+)/; var match = url.match(regex); if (match && match[1]) { directUrl = "https://drive.google.com/thumbnail?id=" + match[1] + "&sz=w1000"; } range.setFormula('=IMAGE("' + directUrl + '")'); } else { range.setValue(url); } }

// Calcula diferencia exacta emulando SIFECHA de Excel
function calcularDiferenciaMesesDias(fechaInicio, fechaFin) {
  if (!fechaInicio || !fechaFin) return "";
  var d1 = new Date(fechaInicio);
  var d2 = new Date(fechaFin);
  if (isNaN(d1.getTime()) || isNaN(d2.getTime())) return "";

  var m1 = d1.getFullYear() * 12 + d1.getMonth();
  var m2 = d2.getFullYear() * 12 + d2.getMonth();
  var meses = m2 - m1;
  var dias = d2.getDate() - d1.getDate();

  if (dias < 0) {
    meses--;
    var ultimoDiaMesAnterior = new Date(d2.getFullYear(), d2.getMonth(), 0).getDate();
    dias += ultimoDiaMesAnterior;
  }

  if (meses < 0) return "";
  return meses + " meses, " + dias + " días";
}

function getFinanzasData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_Perfiles");
  var usersData = dbSheet.getRange("A2:B" + Math.max(2, dbSheet.getLastRow())).getValues();
  var estadoSheet = ss.getSheetByName("FINANZAS_ESTADO");
  var estadoData = estadoSheet ? estadoSheet.getRange("A2:D" + Math.max(2, estadoSheet.getLastRow())).getValues() : [];
  var estadosMap = {};
  for(var i = 0; i < estadoData.length; i++) { if(estadoData[i][0]) estadosMap[estadoData[i][0]] = { base: estadoData[i][1], plan: estadoData[i][2], proximo: estadoData[i][3] }; }
  var pagosSheet = ss.getSheetByName("FINANZAS_PAGOS");
  var pagosData = pagosSheet ? pagosSheet.getRange("A2:I" + Math.max(2, pagosSheet.getLastRow())).getDisplayValues() : [];
  var usuarios = []; var alertas = []; var hoy = new Date(); hoy.setHours(0,0,0,0); var limiteAlerta = new Date(hoy); limiteAlerta.setDate(limiteAlerta.getDate() + 5);
  for(var j = 0; j < usersData.length; j++) {
    var uid = usersData[j][0]; if(!uid) continue;
    var nombre = usersData[j][1] || "Sin Nombre"; var est = estadosMap[uid] || { base: "", plan: "Mensual", proximo: "" };
    usuarios.push({ uid: uid, nombre: nombre, base: est.base ? Utilities.formatDate(new Date(est.base), "GMT-4", "yyyy-MM-dd") : "", plan: est.plan, proximo: est.proximo ? Utilities.formatDate(new Date(est.proximo), "GMT-4", "dd/MM/yyyy") : "Sin pagos" });
    if(est.proximo && est.proximo instanceof Date) { var fProx = new Date(est.proximo); if(fProx < hoy) alertas.push({ nombre: nombre, fecha: Utilities.formatDate(fProx, "GMT-4", "dd/MM/yyyy"), estado: "VENCIDO" }); else if(fProx <= limiteAlerta) alertas.push({ nombre: nombre, fecha: Utilities.formatDate(fProx, "GMT-4", "dd/MM/yyyy"), estado: "PRÓXIMO A VENCER" }); }
  }
  var pagosConFila = [];
  for(var p = 0; p < pagosData.length; p++) { if(pagosData[p][0]) { pagosConFila.push({ fila: p + 2, uid: pagosData[p][0], nombre: pagosData[p][1], fechaPago: pagosData[p][2], mes: pagosData[p][3], anio: pagosData[p][4], plan: pagosData[p][5], metodo: pagosData[p][6], recibo: pagosData[p][7], costo: pagosData[p][8] || "" }); } }
  pagosConFila.reverse(); usuarios.sort(function(a, b) { return a.nombre.localeCompare(b.nombre); });
  return { usuarios: usuarios, alertas: alertas, pagos: pagosConFila };
}

function registrarPago(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var pagosSheet = ss.getSheetByName("FINANZAS_PAGOS");
  var estadoSheet = ss.getSheetByName("FINANZAS_ESTADO");
  pagosSheet.appendRow([payload.uid, payload.nombre, new Date(), payload.mes, payload.anio, payload.plan, payload.metodo, payload.comprobante, payload.costo]);
  var estData = estadoSheet.getRange("A2:A" + Math.max(2, estadoSheet.getLastRow())).getValues();
  var fila = -1;
  for(var i = 0; i < estData.length; i++) { if(estData[i][0] == payload.uid) { fila = i + 2; break; } }
  var fBaseParts = payload.fechaBase.split('-'); var fBase = new Date(fBaseParts[0], fBaseParts[1] - 1, fBaseParts[2]);
  var mesesArray = ["ENERO","FEBRERO","MARZO","ABRIL","MAYO","JUNIO","JULIO","AGOSTO","SEPTIEMBRE","OCTUBRE","NOVIEMBRE","DICIEMBRE"];
  var mesIndex = mesesArray.indexOf(payload.mes); 
  var fProximo;
  if (payload.plan === "Anual") { fProximo = new Date(parseInt(payload.anio) + 1, mesIndex, fBase.getDate()); } else { fProximo = new Date(parseInt(payload.anio), mesIndex + 1, fBase.getDate()); }
  if(fila === -1) { fila = estadoSheet.getLastRow() + 1; estadoSheet.getRange(fila, 1, 1, 4).setValues([[payload.uid, fBase, payload.plan, fProximo]]); } else { estadoSheet.getRange(fila, 2, 1, 3).setValues([[fBase, payload.plan, fProximo]]); }
  return "Pago sincronizado con éxito";
}

    }
    var dbSheet = ss.getSheetByName("DB_Perfiles");
    var correos = [];
    if (dbSheet) {
      var dataU = dbSheet.getDataRange().getValues();
      for (var i = 1; i < dataU.length; i++) {
        if (dataU[i][15] && dataU[i][15] !== "") correos.push(dataU[i][15]);
      }
    }
    var uF = hojaSaaS.getLastRow();
    var cursos = [];
    if (uF >= 3) {
      cursos = hojaSaaS.getRange(3, 1, uF - 2, 10).getValues().map(function(f) { 
        return f.map(function(c) { return (c instanceof Date) ? Utilities.formatDate(c, Session.getScriptTimeZone(), "yyyy-MM-dd") : c; });
      });
    }
    return { cursos: cursos, clientes: correos };
  } catch (e) { return { error: e.toString() }; }
}

function guardarCambioControlSaaS(index, tipo, valor) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
    var fila = parseInt(index) + 3;
    var columnas = { 'curso': 1, 'cliente': 2, 'monto': 3, 'cobrado': 4, 'fecha_suscripcion': 5, 'fecha_pago': 6, 'comprobante': 10 };
    
    if (columnas[tipo]) {
      if (valor === "" && ['fecha_suscripcion', 'fecha_pago'].includes(tipo)) {
         hojaSaaS.getRange(fila, columnas[tipo]).clearContent();
      } else {
         hojaSaaS.getRange(fila, columnas[tipo]).setValue(valor);
      }

      if (tipo === 'cobrado' && valor === true) {
        var fechaHoy = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
        if (!hojaSaaS.getRange(fila, 6).getValue()) { hojaSaaS.getRange(fila, 6).setValue(fechaHoy); }
      }

      if (['cobrado', 'fecha_pago', 'cliente'].includes(tipo)) { actualizarEstadosVencimientoSaaS(); }
      return "Guardado";
    }
    return "Atributo no encontrado";
  } catch (e) { return "Error: " + e.toString(); }
}

function generarSiguienteMesSaaS(index, nuevoMonto) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
    var filaOriginal = parseInt(index) + 3;
    
    var cursoServicio = hojaSaaS.getRange(filaOriginal, 1).getValue();
    var cliente = hojaSaaS.getRange(filaOriginal, 2).getValue();
    var monto = nuevoMonto || hojaSaaS.getRange(filaOriginal, 3).getValue();
    var fechaSuscripcion = hojaSaaS.getRange(filaOriginal, 5).getValue();
    var fechaPagoAnterior = hojaSaaS.getRange(filaOriginal, 6).getValue();
    
    var nuevaFecha = new Date();
    if (fechaPagoAnterior instanceof Date) {
      nuevaFecha = new Date(fechaPagoAnterior);
    } else if (typeof fechaPagoAnterior === 'string' && fechaPagoAnterior.trim() !== '') {
      var parts = fechaPagoAnterior.split('-');
      if(parts.length===3) nuevaFecha = new Date(parts[0], parts[1]-1, parts[2]);
    } else if (fechaSuscripcion instanceof Date) {
      nuevaFecha = new Date(fechaSuscripcion);
    }
    
    nuevaFecha.setMonth(nuevaFecha.getMonth() + 1);
    var nuevaFechaStr = Utilities.formatDate(nuevaFecha, Session.getScriptTimeZone(), "yyyy-MM-dd");
    var fechaSuscripcionStr = (fechaSuscripcion instanceof Date) ? Utilities.formatDate(fechaSuscripcion, Session.getScriptTimeZone(), "yyyy-MM-dd") : fechaSuscripcion;

    hojaSaaS.appendRow([cursoServicio, cliente, monto, false, fechaSuscripcionStr, nuevaFechaStr, "", "", "", ""]);
    return "Siguiente mes generado";
  } catch(e) { return "Error: " + e.toString(); }
}

function actualizarEstadosVencimientoSaaS() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
  if (!hojaSaaS) return;
  var uF = hojaSaaS.getLastRow();
  if (uF < 3) return;

  var rango = hojaSaaS.getRange(3, 1, uF - 2, 10);
  var datos = rango.getValues();
  var hoy = new Date();
  hoy.setHours(0,0,0,0);
  var nuevosEstados = [];

  for (var i = 0; i < datos.length; i++) {
    var fila = datos[i]; var cobrado = fila[3]; var fechaPago = fila[5]; var estK = fila[6]; var estL = fila[7]; var estM = fila[8]; 

    if (fechaPago instanceof Date && cobrado !== true) {
      var fechaComp = new Date(fechaPago);
      fechaComp.setHours(0,0,0,0);
      var difDias = Math.floor((hoy - fechaComp) / (1000 * 60 * 60 * 24));

      if (difDias >= 0 && (estK === "" || estK === null)) estK = "VENCIDO";
      if (difDias >= 7 && (estL === "" || estL === null)) estL = "VENCIDO 7D";
      if (difDias >= 15 && (estM === "" || estM === null)) estM = "VENCIDO 15D";
    }

    if (cobrado === true) {
      if (estK.toString().includes("VENCIDO")) estK = "";
      if (estL.toString().includes("VENCIDO")) estL = "";
      if (estM.toString().includes("VENCIDO")) estM = "";
    }
    nuevosEstados.push([estK, estL, estM]);
  }
  hojaSaaS.getRange(3, 7, nuevosEstados.length, 3).setValues(nuevosEstados);
}

function ejecutarEnvioCobroSaaS() {
  actualizarEstadosVencimientoSaaS();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
  var hojaScripts = ss.getSheetByName("SCRIPTS DE COBRO");
  if (!hojaSaaS || !hojaScripts) return "No se encontraron las tablas base.";

  var datos = hojaSaaS.getDataRange().getValues();
  var scripts = [
    hojaScripts.getRange("B6").getValue(),
    hojaScripts.getRange("B14").getValue(),
    hojaScripts.getRange("B22").getValue()
  ];
  var conteo = 0;

  for (var i = 2; i < datos.length; i++) {
    var fila = datos[i]; var msgIndex = -1; var colARecordar = -1;
    if (fila[6] === "ENVIAR" || fila[6] === "VENCIDO") { msgIndex = 0; colARecordar = 7; }
    else if (fila[7] === "ENVIAR 7D" || fila[7] === "VENCIDO 7D") { msgIndex = 1; colARecordar = 8; }
    else if (fila[8] === "ENVIAR 15D" || fila[8] === "VENCIDO 15D") { msgIndex = 2; colARecordar = 9; }

    if (msgIndex !== -1 && scripts[msgIndex]) {
      var email = fila[1] ? fila[1].toString().trim() : "";
      if (email && email.includes("@")) {
        var cuerpo = scripts[msgIndex].replace("[Nombre]", email).replace("[Proyecto]", fila[0]).replace("[Monto]", fila[2]);
        try {
          MailApp.sendEmail(email, "⚠️ Recordatorio de Pago SaaS: " + (fila[0] || "Suscripción"), cuerpo);
          hojaSaaS.getRange(i + 1, colARecordar).setValue("ENVIADO");
          conteo++;
        } catch (e) { Logger.log("Error envío SaaS: " + e.message); }
      }
    }
  }
  return "Se enviaron " + conteo + " recordatorios de cobro SaaS.";
}

function obtenerDatosReporteAnualSaaS(anio) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
    var mesesTotales = new Array(12).fill(0);
    var totalAnio = 0;

    if (hojaSaaS && hojaSaaS.getLastRow() >= 3) {
      var datosC = hojaSaaS.getRange(3, 1, hojaSaaS.getLastRow() - 2, 6).getValues();
      datosC.forEach(function(fila) {
        var monto = parseFloat(fila[2]) || 0;
        var cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
        var fecha = fila[5]; 
        if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
          if (fecha.getFullYear() == anio) {
            mesesTotales[fecha.getMonth()] += monto;
            totalAnio += monto;
          }
        }
      });
    }
    return { meses: mesesTotales, total: totalAnio };
  } catch(e) { return { error: e.toString() }; }
}

function obtenerDatosReporteClientesSaaS(anio, mesTexto) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaSaaS = ss.getSheetByName("CONTROL_SAAS");
    var mesesArray = ["ENERO", "FEBRERO", "MARZO", "ABRIL", "MAYO", "JUNIO", "JULIO", "AGOSTO", "SEPTIEMBRE", "OCTUBRE", "NOVIEMBRE", "DICIEMBRE"];
    var mesIndex = mesesArray.indexOf(mesTexto.toUpperCase());
    var clientesMap = {};

    if (hojaSaaS && hojaSaaS.getLastRow() >= 3) {
      var datosC = hojaSaaS.getRange(3, 1, hojaSaaS.getLastRow() - 2, 6).getValues();
      datosC.forEach(function(fila) {
        var cliente = fila[1] || "Sin Usuario";
        var monto = parseFloat(fila[2]) || 0;
        var cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
        var fecha = fila[5]; 
        if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
          if (fecha.getFullYear() == anio && fecha.getMonth() == mesIndex) {
            if(!clientesMap[cliente]) clientesMap[cliente] = 0;
            clientesMap[cliente] += monto;
          }
        }
      });
    }
    return { labels: Object.keys(clientesMap), valores: Object.values(clientesMap) };
  } catch(e) { return { error: e.toString() }; }
}
