function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000); 
    
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var userId = data.userId;
    var payload = data.datos || {};

    if (!userId) return ContentService.createTextOutput(JSON.stringify({ error: "Falta userId" })).setMimeType(ContentService.MimeType.JSON);

    var resultado = "";
    switch (action) {
      // MÉTODOS BÁSICOS DE USUARIO
      case "getDashboardData": resultado = getDashboardData(userId, payload.email); break;
      case "saveWelcomeData": resultado = saveWelcomeData(payload, userId); break;
      case "guardarEscalon": resultado = guardarEscalon(payload, userId); break;
      case "guardarFotoEscalonBackend": resultado = guardarFotoEscalonBackend(payload, userId); break;

function registrarTelefono(payload, userId) {
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  dbSheet.getRange("R1").setValue("TELEFONO"); // Asegurar el título en R1
  dbSheet.getRange("S1").setValue("FECHA REGISTRO"); dbSheet.getRange("T1").setValue("ROL");
  var uids = dbSheet.getRange("A2:A").getValues();
  var targetRow = -1;
  for (var i = 0; i < uids.length; i++) { 
    if (uids[i][0] == userId) { targetRow = i + 2; break; } 
  }
  
  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    var estadoInicial = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    dbSheet.getRange(targetRow, 15).setValue(estadoInicial); 
    dbSheet.getRange(targetRow, 19).setValue(new Date()); // FECHA REGISTRO
    dbSheet.getRange(targetRow, 20).setValue((userId === ADMIN_UID) ? "Administrador" : "Cliente"); // ROL
    if (payload.email) dbSheet.getRange(targetRow, 16).setValue(payload.email); 
  }
  
  dbSheet.getRange(targetRow, 18).setValue(payload.telefono); // Columna R
  return "Teléfono guardado";
}

function getUserSpreadsheet(userId, email, nombre) {
  var centralSs = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = centralSs.getSheetByName("DB_Perfiles");
  var data = dbSheet.getRange("A2:T").getValues(); 
  var targetRow = -1;
  var sheetId = "";
  var estado = "Autorizado";
  var rol = "Cliente";

  for (var i = 0; i < data.length; i++) {
    if (data[i][0] == userId) {
      targetRow = i + 2;
      var reU = leerRolEstado(data[i]); estado = reU.estado; rol = reU.rol; 
      sheetId = data[i][16]; // Col Q
      
      if (email && data[i][15] === "") dbSheet.getRange(targetRow, 16).setValue(email);
      break;
    }
  }

  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    estado = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    rol = (userId === ADMIN_UID) ? "Administrador" : "Cliente";
    dbSheet.getRange(targetRow, 15).setValue(estado); 
    dbSheet.getRange(targetRow, 19).setValue(new Date()); // FECHA REGISTRO
    dbSheet.getRange(targetRow, 20).setValue(rol); // ROL
    if (email) dbSheet.getRange(targetRow, 16).setValue(email); 
  }

  if (!sheetId || sheetId === "") {
    if (userId === ADMIN_UID) {
      sheetId = TEMPLATE_ID;
      dbSheet.getRange(targetRow, 17).setValue(sheetId);
    } else {
      var template = DriveApp.getFileById(TEMPLATE_ID);
      var targetFolder = DriveApp.getFolderById(FOLDER_ID);
      var copy = template.makeCopy("Fitness Pro - " + (email || userId), targetFolder);
      sheetId = copy.getId();
      dbSheet.getRange(targetRow, 17).setValue(sheetId);
    }
  }

  return { ss: SpreadsheetApp.openById(sheetId), estado: estado, rol: rol, esNuevo: (sheetId === "") };
}

function parseFloatSafe(val) { if (val === "" || val === undefined || val === null) return ""; var str = String(val).replace(',', '.'); var num = parseFloat(str); return isNaN(num) ? "" : num; }
function readImageUrl(sheet, cellStr) { var formula = sheet.getRange(cellStr).getFormula(); if (formula && formula.toUpperCase().indexOf('=IMAGE') !== -1) { var match = formula.match(/=IMAGE\("([^"]+)"/i); if (match) return match[1]; } return sheet.getRange(cellStr).getDisplayValue(); }
function writeImageUrl(sheet, cellStr, url) { var range = sheet.getRange(cellStr); if (url && url.toString().startsWith("http")) { var directUrl = url; var regex = /drive\.google\.com\/(?:file\/d\/|open\?id=)([-_a-zA-Z0-9]+)/; var match = url.match(regex); if (match && match[1]) { directUrl = "https://drive.google.com/thumbnail?id=" + match[1] + "&sz=w1000"; } range.setFormula('=IMAGE("' + directUrl + '")'); } else { range.setValue(url); } }

// Calcula diferencia exacta emulando SIFECHA de Excel
function calcularDiferenciaMesesDias(fechaInicio, fechaFin) {
  if (!fechaInicio || !fechaFin) return "";
  var d1 = new Date(fechaInicio);
  var d2 = new Date(fechaFin);
  if (isNaN(d1.getTime()) || isNaN(d2.getTime())) return "";

  var m1 = d1.getFullYear() * 12 + d1.getMonth();
  var m2 = d2.getFullYear() * 12 + d2.getMonth();
  var meses = m2 - m1;
  var dias = d2.getDate() - d1.getDate();

  if (dias < 0) {
    meses--;
    var ultimoDiaMesAnterior = new Date(d2.getFullYear(), d2.getMonth(), 0).getDate();
    dias += ultimoDiaMesAnterior;
  }

  if (meses < 0) return "";
  return meses + " meses, " + dias + " días";
}

function getDashboardData(userId, email) {
  var cache = CacheService.getScriptCache();
  var cacheKey = "dash_" + userId;
  var esSuperAdmin = (userId === ADMIN_UID);

  if (!esSuperAdmin) { var cachedData = cache.get(cacheKey); if (cachedData) return JSON.parse(cachedData); }

  var userEnv = getUserSpreadsheet(userId, email, "");
  var esAdminRegular = (userEnv.rol === "Administrador" && userEnv.estado === "Autorizado");
  var tieneRolAdministrador = esSuperAdmin || esAdminRegular;
  
  if (userEnv.estado === "Pendiente" && !esSuperAdmin) { return { isPendiente: true }; }
  if (userEnv.estado === "Bloqueado" && !esSuperAdmin) { return { isBloqueado: true }; }

  var pSheet = userEnv.ss.getSheetByName("Perfil");
  var pRange = pSheet.getRange(1, 1, 175, 15);
  
  // Extraemos objetos Date nativos para el cálculo y DisplayValues para la APP
  var pRawObjects = pRange.getValues(); 
  var pFormulas = pRange.getFormulas();
  var pValues = pRange.getDisplayValues();

  function fastReadImage(row, col) { var f = pFormulas[row-1][col-1]; if (f && f.toUpperCase().indexOf('=IMAGE') !== -1) { var match = f.match(/=IMAGE\("([^"]+)"/i); if (match) return match[1]; } return pValues[row-1][col-1]; }
  function fastVal(row, col) { return pValues[row-1][col-1]; }

  var diarioSheet = userEnv.ss.getSheetByName("Diario"); 
  var calcSheet = userEnv.ss.getSheetByName("Calculadora") || userEnv.ss.getSheetByName("Calculadoras");
  var centralSs = SpreadsheetApp.getActiveSpreadsheet();
  var recetasSheet = centralSs.getSheetByName("RECETAS") || centralSs.getSheetByName("Recetas"); 
  var appLogoUrl = readImageUrl(centralSs.getSheetByName("DB_Perfiles"), "N2");

  var DEF_ESC = "https://i.postimg.cc/XvjwD97P/Tu-foto-aqui.jpg"; var DEF_FRE = "https://i.postimg.cc/dVdHKRY5/FRENTE.jpg"; var DEF_ESP = "https://i.postimg.cc/7ZY3KgL7/ESPALDA.jpg"; var DEF_PD = "https://i.postimg.cc/CKRjgSkk/PERFIL-DERECHO.jpg"; var DEF_PI = "https://i.postimg.cc/RhLtRF5D/PERFIL-IZQUIERDO.jpg";

  var nombre = fastVal(2, 1) || "Usuario";
  var fechaReg = fastVal(2, 3);
  var pesoInicial = fastVal(10, 1);
  var talla = fastVal(7, 1);
  var pesoIdeal = fastVal(13, 1); 
  var edad = fastVal(4, 1) || 30;
  var fechaNac = fastVal(19, 1); 
  var pesoActual = fastVal(31, 11); 
  var deficit = diarioSheet ? diarioSheet.getRange("K382").getDisplayValue() : fastVal(19, 3);
  var pesoObjetivo = fastVal(16, 1);
  var genero = calcSheet ? calcSheet.getRange("D2").getDisplayValue() : "Hombre";
  var imcVal = calcSheet ? calcSheet.getRange("D7").getDisplayValue() : "";
  var avatarUrl = fastReadImage(21, 1);
  var evolucionHoy = fastVal(27, 11);

  // AUTO-SANACIÓN DEL EXCEL: Si eliminaste la fórmula, el script detecta las celdas L vacías y las imprime automáticamente
  var c2Date = pRawObjects[1][2]; // C2 Object Date
  var filasFechas = [42, 53, 64, 75, 86, 97, 108, 119, 130, 141, 152, 163]; var escalonesData = [];
  
  for (var i = 0; i < filasFechas.length; i++) { 
    var filaF = filasFechas[i]; 
    var escUrl = fastReadImage(34 + (i * 11), 14); 
    
    var kDateObj = pRawObjects[filaF - 1][10]; // Columna K (índice 10)
    var lValorActual = pRawObjects[filaF - 1][11]; // Columna L (índice 11)

    var diasFormateados = "";
    if (c2Date instanceof Date && kDateObj instanceof Date) {
      diasFormateados = calcularDiferenciaMesesDias(c2Date, kDateObj);
    }
    
    // Si la celda en Excel está vacía o distinta, la re-escribimos directamente en el Spreadsheet
    if (diasFormateados !== "" && lValorActual !== diasFormateados) {
       pSheet.getRange("L" + filaF).setValue(diasFormateados);
    }

    escalonesData.push({ 
      estado: fastVal(filaF - 4, 11), 
      fecha: fastVal(filaF, 11), 
      dias: diasFormateados || lValorActual, 
      foto: (escUrl && escUrl.toString().startsWith("http")) ? escUrl : DEF_ESC 
    }); 
  }

  var medidasMatriz = pSheet.getRange("C2:N18").getDisplayValues(); var mesesData = [];
  for (var m = 1; m <= 12; m++) { var cI = m - 1; var fFM = 23 + ((m - 1) * 11); var uF = fastReadImage(fFM, 2); var uE = fastReadImage(fFM, 4); var uPD = fastReadImage(fFM, 6); var uPI = fastReadImage(fFM, 8); mesesData.push({ mes: m, fechaMed: medidasMatriz[0][cI], pesoColB: medidasMatriz[1][cI], difObj: medidasMatriz[2][cI], med: { cuello: medidasMatriz[3][cI], hombro: medidasMatriz[4][cI], pecho: medidasMatriz[5][cI], cintura: medidasMatriz[6][cI], brazoI: medidasMatriz[7][cI], brazoD: medidasMatriz[8][cI], anteI: medidasMatriz[9][cI], anteD: medidasMatriz[10][cI], circAb: medidasMatriz[11][cI], musloI: medidasMatriz[12][cI], musloD: medidasMatriz[13][cI], pantI: medidasMatriz[14][cI], pantD: medidasMatriz[15][cI], cadera: medidasMatriz[16][cI] }, fotos: { frente: (uF && uF.toString().startsWith("http")) ? uF : DEF_FRE, espalda: (uE && uE.toString().startsWith("http")) ? uE : DEF_ESP, perfilDer: (uPD && uPD.toString().startsWith("http")) ? uPD : DEF_PD, perfilIzq: (uPI && uPI.toString().startsWith("http")) ? uPI : DEF_PI } }); }

  var diarioDataList = []; 
  if (diarioSheet) {
    var lastRowD = Math.min(diarioSheet.getLastRow(), 381);
    if (lastRowD >= 2) { 
      var dValues = diarioSheet.getRange(2, 1, lastRowD - 1, 11).getDisplayValues(); 
      for (var d = 0; d < dValues.length; d++) { 
        diarioDataList.push({ fila: d + 2, fecha: dValues[d][0], peso: dValues[d][1], dif: dValues[d][2], progreso: dValues[d][3], energia: dValues[d][4], caminata: dValues[d][5], agua: dValues[d][6], presion: dValues[d][7], presionStat: dValues[d][8], notas: dValues[d][9] }); 
      } 
    } 
  }

  var fotosCalc = { img1: "", img2: "", img3: "", img4: "", img5: "" };
  if (calcSheet) { 
    fotosCalc.img1 = readImageUrl(calcSheet, "B2"); 
    fotosCalc.img2 = readImageUrl(calcSheet, "E1"); 
    fotosCalc.img3 = readImageUrl(calcSheet, "F1"); 
    fotosCalc.img4 = readImageUrl(calcSheet, "G13"); 
    fotosCalc.img5 = readImageUrl(calcSheet, "G1"); 
  }

  var recetasList = [];
  if (recetasSheet) { var lastRowR = recetasSheet.getLastRow(); if (lastRowR > 1) { var recRaw = recetasSheet.getRange(2, 1, lastRowR - 1, 6).getDisplayValues(); var recFormulas = recetasSheet.getRange(2, 6, lastRowR - 1, 1).getFormulas(); for(var r = 0; r < recRaw.length; r++) { if(!recRaw[r][0]) continue; var picUrl = recRaw[r][5]; var form = recFormulas[r][0]; if (form && form.toUpperCase().indexOf('=IMAGE') !== -1) { var match = form.match(/=IMAGE\("([^"]+)"/i); if (match) picUrl = match[1]; } recetasList.push({ fila: r + 2, nombre: recRaw[r][0], categoria: (recRaw[r][1] || "").toUpperCase().trim(), calorias: recRaw[r][2], ingredientes: recRaw[r][3], preparacion: recRaw[r][4], fotoRaw: picUrl }); } } }

  var estadoSheet = centralSs.getSheetByName("FINANZAS_ESTADO");
  var pagosSheet = centralSs.getSheetByName("FINANZAS_PAGOS");
  var finanzasUser = { plan: "Sin plan", proximo: "--", estado: "Sin registro" };
  var historialPagos = [];
  var hoyF = new Date(); hoyF.setHours(0,0,0,0);

  if (estadoSheet) {
    var eData = estadoSheet.getDataRange().getValues();
    for (var e = 1; e < eData.length; e++) {
      if (eData[e][0] === userId) {
        finanzasUser.plan = eData[e][2] || "Sin plan";
        var fProx = eData[e][3];
        if (fProx && fProx instanceof Date) {
          finanzasUser.proximo = Utilities.formatDate(fProx, "GMT-4", "dd/MM/yyyy");
          finanzasUser.estado = (fProx < hoyF) ? "VENCIDO" : "AL DÍA";
        }
        break;
      }
    }
  }
  
  if (pagosSheet) {
    var pData = pagosSheet.getRange("A2:I" + Math.max(2, pagosSheet.getLastRow())).getDisplayValues();
    for (var p = 0; p < pData.length; p++) {
      if (pData[p][0] === userId) { historialPagos.push({ fecha: (pData[p][2]||"").substring(0,10), periodo: pData[p][3] + " " + pData[p][4], plan: pData[p][5], metodo: pData[p][6], recibo: pData[p][7], costo: pData[p][8] || "" }); }
    }
    historialPagos.reverse(); 
  }

  var responsePayload = { 
    nombre: nombre, fechaReg: fechaReg, pesoInicial: pesoInicial, talla: talla, pesoIdeal: pesoIdeal, edad: edad, 
    fechaNac: fechaNac, peso: pesoActual, deficit: deficit, pesoObjetivo: pesoObjetivo, genero: genero, imc: imcVal, 
    evolucion: evolucionHoy, escalones: escalonesData, meses: mesesData, diario: diarioDataList, calcFotos: fotosCalc, 
    recetas: recetasList, 
    presionActual: diarioSheet ? diarioSheet.getRange("H382").getDisplayValue() : "", 
    presionEstado: diarioSheet ? diarioSheet.getRange("I382").getDisplayValue() : "", 
    ultimaCaminata: diarioSheet ? diarioSheet.getRange("F382").getDisplayValue() : "", 
    ultimoAgua: diarioSheet ? diarioSheet.getRange("G382").getDisplayValue() : "", 
    avatarUrl: avatarUrl, appLogoUrl: appLogoUrl, isAdmin: tieneRolAdministrador, 
    isSuperAdmin: esSuperAdmin, finanzasCli: finanzasUser, historialPagos: historialPagos 
  };

  try { if (!esSuperAdmin) { var jsonStr = JSON.stringify(responsePayload); if (jsonStr.length < 90000) { cache.put(cacheKey, jsonStr, 300); } } } catch (errCache) {}
  return responsePayload;
}

function saveWelcomeData(datos, userId) {
  var userEnv = getUserSpreadsheet(userId, null, datos.nombre); var sheet = userEnv.ss.getSheetByName("Perfil"); var calcSheet = userEnv.ss.getSheetByName("Calculadora") || userEnv.ss.getSheetByName("Calculadoras");
  if (datos.nombre) sheet.getRange("A2").setValue(datos.nombre); if (datos.fechaReg) sheet.getRange("C2").setValue(datos.fechaReg); if (datos.fecha) sheet.getRange("A19").setValue(datos.fecha); if (datos.pesoIni) sheet.getRange("A10").setValue(parseFloatSafe(datos.pesoIni));
  var tallaNum = parseFloatSafe(datos.talla); if (tallaNum !== "") { sheet.getRange("A7").setValue(tallaNum); if (calcSheet) calcSheet.getRange("D3").setValue(tallaNum); }
  if (datos.pesoObj !== undefined) sheet.getRange("A16").setValue(parseFloatSafe(datos.pesoObj)); 
  if (datos.avatarUrl !== undefined) writeImageUrl(sheet, "A21", datos.avatarUrl); 
  var generoUser = datos.genero || "Hombre"; if (datos.genero && calcSheet) calcSheet.getRange("D2").setValue(generoUser);
  if (tallaNum !== "" && calcSheet) { var tallaCm = tallaNum * 100; var pesoIdealCalc = (generoUser === 'Hombre') ? (tallaCm - 100 - ((tallaCm - 150) / 4)) : (tallaCm - 100 - ((tallaCm - 150) / 2.5)); var pIdArr = Math.round(pesoIdealCalc * 10) / 10; calcSheet.getRange("D6").setValue(pIdArr); sheet.getRange("A13").setValue(pIdArr); }
  restaurarFormulasPerfil(sheet); actualizarDBPerfiles(userId, sheet, datos.nombre); CacheService.getScriptCache().remove("dash_" + userId);
  return "¡Perfil configurado!";
}

function guardarEscalon(payload, userId) { 
  var userEnv = getUserSpreadsheet(userId); 
  var sheet = userEnv.ss.getSheetByName("Perfil"); 
  var filasFechas = [42, 53, 64, 75, 86, 97, 108, 119, 130, 141, 152, 163]; 
  var index = payload.numero - 1;
  
  sheet.getRange("K" + (filasFechas[index] - 4)).setValue(payload.estado); 
  sheet.getRange("K" + filasFechas[index]).setValue(payload.fecha); 
  
  // Imprimir físicamente en Excel la diferencia en columna L
  SpreadsheetApp.flush(); 
  var c2Val = sheet.getRange("C2").getValue();
  var kVal = sheet.getRange("K" + filasFechas[index]).getValue();
  
  if (c2Val instanceof Date && kVal instanceof Date) {
    sheet.getRange("L" + filasFechas[index]).setValue(calcularDiferenciaMesesDias(c2Val, kVal));
  } else {
    sheet.getRange("L" + filasFechas[index]).setValue("");
  }
  
  CacheService.getScriptCache().remove("dash_" + userId); 
  return "Guardado"; 
}

// GESTIÓN DE USUARIOS ADMIN (modelo Control Freelancer: UID, Cliente, Correo, Teléfono, Fecha registro, Rol, Estatus)
// Col O = ESTATUS (Pendiente / Autorizado / Bloqueado) · Col R = TELÉFONO · Col S = FECHA REGISTRO · Col T = ROL (Cliente / Administrador)
// Compatibilidad: el valor antiguo "Admin" en la col O se lee como Rol=Administrador + Estatus=Autorizado.
// ==========================================
function leerRolEstado(fila) {
  var estado = fila[14] || "Autorizado";
  var rol = fila[19] || "Cliente";
  if (estado === "Admin") { estado = "Autorizado"; rol = "Administrador"; }
  return { rol: rol, estado: estado };
}

function getUsersList() {
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var rango = dbSheet.getRange("A2:T" + Math.max(2, dbSheet.getLastRow()));
  var disp = rango.getDisplayValues();
  var vals = rango.getValues();
  var users = []; var rellenados = 0;
  for (var i = 0; i < disp.length; i++) {
    if (disp[i][0] === "") continue;
    var uid = disp[i][0];
    var re = leerRolEstado(vals[i]);
    var fecha = aIsoFecha(vals[i][18]);
    // Usuarios antiguos sin fecha de registro: se toma la fecha de creación de su hoja (máx. 15 por consulta).
    if (!fecha && uid !== ADMIN_UID && disp[i][16] && rellenados < 15) {
      try { var creada = DriveApp.getFileById(disp[i][16]).getDateCreated(); dbSheet.getRange(i + 2, 19).setValue(creada); fecha = aIsoFecha(creada); rellenados++; } catch (e) {}
    }
    users.push({ uid: uid, nombre: disp[i][1], email: disp[i][15], telefono: disp[i][17], fechaRegistro: fecha, rol: re.rol, estado: re.estado, protegido: (uid === ADMIN_UID) });
  }
  return users;
}

function actualizarUsuario(p, quien) {
  var uid = String(p.uid || "");
  if (!uid) throw "Falta el usuario.";
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var uids = dbSheet.getRange("A2:A" + Math.max(2, dbSheet.getLastRow())).getValues();
  for (var i = 0; i < uids.length; i++) {
    if (uids[i][0] == uid) {
      var fila = i + 2;
      dbSheet.getRange(fila, 18).setValue(String(p.telefono || ""));
      // Ni el súper admin ni tu propia cuenta pueden cambiar de rol o estatus.
      if (uid !== ADMIN_UID && uid !== quien) {
        var rol = (p.rol === "Administrador") ? "Administrador" : "Cliente";
        var estado = (["Pendiente", "Autorizado", "Bloqueado"].indexOf(p.estado) >= 0) ? p.estado : "Pendiente";
        dbSheet.getRange(fila, 15).setValue(estado);
        dbSheet.getRange(fila, 20).setValue(rol);
      }
      CacheService.getScriptCache().remove("dash_" + uid);
      return "Usuario actualizado";
    }
  }
  throw "Usuario no encontrado.";
}

function eliminarUser(uid, quien) {
  if (uid === quien || uid === ADMIN_UID) throw "No puedes eliminar tu propia cuenta de administrador.";
  var ss = SpreadsheetApp.getActiveSpreadsheet(); var dbSheet = ss.getSheetByName("DB_Perfiles");
  var uids = dbSheet.getRange("A2:A" + Math.max(2, dbSheet.getLastRow())).getValues();
  for (var i = 0; i < uids.length; i++) {
    if (uids[i][0] == uid) {
      var sheetId = dbSheet.getRange(i + 2, 17).getValue();
      if (sheetId) { try { DriveApp.getFileById(sheetId).setTrashed(true); } catch (e) {} }
      dbSheet.deleteRow(i + 2);
      CacheService.getScriptCache().remove("dash_" + uid);
      return "Usuario eliminado correctamente";
    }
  }
  throw "Usuario no encontrado.";
}

function getFinanzasData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_Perfiles");
  var usersData = dbSheet.getRange("A2:T" + Math.max(2, dbSheet.getLastRow())).getValues();

  var hojaEstado = getHojaEstado();
  var eData = hojaEstado.getLastRow() >= 2 ? hojaEstado.getRange(2, 1, hojaEstado.getLastRow() - 1, 4).getValues() : [];
  var estados = {};
  for (var e = 0; e < eData.length; e++) { if (eData[e][0]) estados[eData[e][0]] = { plan: eData[e][2], proximo: aIsoFecha(eData[e][3]) }; }

  var nombres = {}; var usuarios = [];
  for (var j = 0; j < usersData.length; j++) {
    var uid = usersData[j][0]; if (!uid) continue;
    var correo = String(usersData[j][15] || ""); var nombre = String(usersData[j][1] || "");
    nombres[uid] = { correo: correo, nombre: nombre };
    if (leerRolEstado(usersData[j]).estado !== "Autorizado") continue; // solo usuarios autorizados
    var est = estados[uid];
    usuarios.push({ uid: uid, correo: correo, nombre: nombre, plan: est ? est.plan : "", proximo: est ? est.proximo : "" });
  }
  usuarios.sort(function(a, b) { return (a.correo || a.nombre).toLowerCase().localeCompare((b.correo || b.nombre).toLowerCase()); });

  var hojaPagos = getHojaPagos();
  var pData = hojaPagos.getLastRow() >= 2 ? hojaPagos.getRange(2, 1, hojaPagos.getLastRow() - 1, 11).getValues() : [];
  var pagos = [];
  for (var p = pData.length - 1; p >= 0 && pagos.length < 100; p--) { // lo más reciente arriba
    var f = pData[p]; if (!f[0]) continue;
    var info = nombres[f[0]] || { correo: "", nombre: "" };
    var mes = String(f[3] || ""); mes = mes ? mes.charAt(0).toUpperCase() + mes.slice(1).toLowerCase() : "";
    pagos.push({ fecha: aIsoFecha(f[2]) || String(f[2] || "").substring(0, 10), correo: String(f[10] || info.correo || ""), nombre: info.nombre || String(f[1] || ""), periodo: (mes + " " + (f[4] || "")).trim(), plan: f[5], metodo: f[6], monto: f[8], recibo: f[7] });
  }
  return { usuarios: usuarios, pagos: pagos };
}

// ---------- Fechas (mismas reglas que lib/fechas.ts de Freelancer) ----------
function aIsoFecha(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd");
  if (typeof v === "string") { var m = v.match(/^(\d{4})-(\d{2})-(\d{2})/); if (m) return m[1] + "-" + m[2] + "-" + m[3]; }
  return "";
}
function parseFechaIso(iso) { var m = String(iso || "").match(/^(\d{4})-(\d{2})-(\d{2})/); return m ? new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])) : null; }
function isoDesdeFecha(d) { var mm = ("0" + (d.getMonth() + 1)).slice(-2); var dd = ("0" + d.getDate()).slice(-2); return d.getFullYear() + "-" + mm + "-" + dd; }
function hoyIso(hoyCliente) { return /^\d{4}-\d{2}-\d{2}$/.test(hoyCliente || "") ? hoyCliente : Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"); }
function diasDesdeIso(iso, hoy) { var f = parseFechaIso(iso), r = parseFechaIso(hoy); if (!f || !r) return null; return Math.round((r.getTime() - f.getTime()) / 86400000); }
// 30-ago + 1 mes = 30-sep y 31-ago + 1 mes = 30-sep (no desborda al mes siguiente)
function sumarMesesIso(iso, meses) {
  var f = parseFechaIso(iso) || new Date(); var dia = f.getDate();
  f.setDate(1); f.setMonth(f.getMonth() + meses);
  var ultimo = new Date(f.getFullYear(), f.getMonth() + 1, 0).getDate();
  f.setDate(Math.min(dia, ultimo));
  return isoDesdeFecha(f);
}
// Mensual: mismo día del mes siguiente · Anual: la misma fecha
function fechaPagoDeCiclo(fsusc, tipo) { return tipo === "Anual" ? fsusc : sumarMesesIso(fsusc, 1); }
function siguienteCiclo(fsusc, fpago, tipo, hoy) {
  var inicio = (tipo === "Anual") ? sumarMesesIso(fsusc || fpago || hoy, 12) : (fpago || (fsusc ? sumarMesesIso(fsusc, 1) : hoy));
  return { fecha_suscripcion: inicio, fecha_pago: fechaPagoDeCiclo(inicio, tipo) };
}

function filaDesdeValores(f, nFila) {
  return {
    fila: nFila, id: String(f[14] || ""), plan: String(f[0] || ""), correo: String(f[1] || "").trim(),
    monto: parseFloat(f[2]) || 0, cobrado: (f[3] === true || String(f[3]).toUpperCase() === "TRUE"),
    fecha_suscripcion: aIsoFecha(f[4]), fecha_pago: aIsoFecha(f[5]),
    marcas: [String(f[6] || ""), String(f[7] || ""), String(f[8] || "")],
    comprobante_url: String(f[9] || ""), metodo_pago: String(f[12] || ""), tipo_suscripcion: (f[13] === "Anual" ? "Anual" : "Mensual")
  };
}

// Lee todas las filas con datos. Le asigna ID a las que no lo tengan (en un solo bloque).
function leerFilasControl(hoja) {
  var ultima = hoja.getLastRow();
  if (ultima < CONTROL_PRIMERA_FILA) return [];
  var vals = hoja.getRange(CONTROL_PRIMERA_FILA, 1, ultima - CONTROL_PRIMERA_FILA + 1, CONTROL_COLS).getValues();
  var filas = []; var faltanIds = false; var colIds = [];
  for (var i = 0; i < vals.length; i++) {
    var f = vals[i];
    var vacia = (f[0] === "" && f[1] === "" && f[2] === "" && f[4] === "" && f[5] === "" && f[9] === "" && f[12] === "");
    if (!vacia && f[14] === "") { f[14] = Utilities.getUuid(); faltanIds = true; }
    colIds.push([f[14]]);
    if (!vacia) filas.push(filaDesdeValores(f, CONTROL_PRIMERA_FILA + i));
  }
  if (faltanIds) hoja.getRange(CONTROL_PRIMERA_FILA, 15, colIds.length, 1).setValues(colIds);
  return filas;
}

function buscarFilaControl(hoja, id) {
  var filas = leerFilasControl(hoja);
  for (var i = 0; i < filas.length; i++) { if (filas[i].id === id) return filas[i]; }
  return null;
}

function primeraFilaLibreControl(hoja) {
  var ultima = hoja.getLastRow();
  if (ultima < CONTROL_PRIMERA_FILA) return CONTROL_PRIMERA_FILA;
  var vals = hoja.getRange(CONTROL_PRIMERA_FILA, 1, ultima - CONTROL_PRIMERA_FILA + 1, CONTROL_COLS).getValues();
  for (var i = 0; i < vals.length; i++) {
    var f = vals[i];
    if (f[0] === "" && f[1] === "" && f[2] === "" && f[4] === "" && f[5] === "" && f[9] === "" && f[12] === "" && f[14] === "") return CONTROL_PRIMERA_FILA + i;
  }
  return ultima + 1;
}

// Vence Hoy / 7 Días / 15 Días: "ENVIADO" si ya se mandó, "VENCIDO…" si ya pasó el plazo, "" si nada.
function estadosVencimientoFila(f, hoy) {
  var dias = f.cobrado ? null : diasDesdeIso(f.fecha_pago, hoy);
  return ETAPAS_COBRO.map(function(et, i) {
    if (f.marcas[i] === "ENVIADO") return "ENVIADO";
    if (dias !== null && dias >= et.dias) return et.etiqueta;
    return "";
  });
}
// Se envía una sola etapa por ejecución y en orden: hoy → 7D → 15D.
function etapaPendienteFila(f, hoy) {
  if (f.cobrado) return -1;
  var dias = diasDesdeIso(f.fecha_pago, hoy); if (dias === null) return -1;
  for (var i = 0; i < ETAPAS_COBRO.length; i++) { if (dias >= ETAPAS_COBRO[i].dias && f.marcas[i] !== "ENVIADO") return i; }
  return -1;
}

// Deja en la hoja el mismo texto que ve el admin (VENCIDO / ENVIADO) para poder leerla a mano.
function actualizarEstadosVencimientoSaaS(hoy) {
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);
  if (!filas.length) return;
  var primera = filas[0].fila; var ultima = filas[filas.length - 1].fila;
  var rango = hoja.getRange(primera, 7, ultima - primera + 1, 3); var vals = rango.getValues();
  for (var i = 0; i < filas.length; i++) {
    var est = estadosVencimientoFila(filas[i], hoy); var k = filas[i].fila - primera;
    vals[k] = est;
  }
  rango.setValues(vals);
}

function listarCorreosUsuarios() {
  var mapa = mapaUsuarios(true); var lista = [];
  for (var c in mapa) { lista.push(mapa[c].correo); }
  lista.sort(function(a, b) { return a.toLowerCase().localeCompare(b.toLowerCase()); });
  return lista;
}

// correo (minúsculas) → { uid, nombre, correo }
function mapaUsuarios(forzar) {
  if (_mapaUsuarios && !forzar) return _mapaUsuarios;
  var db = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var data = db.getRange("A2:P" + Math.max(2, db.getLastRow())).getValues(); var mapa = {};
  for (var i = 0; i < data.length; i++) {
    var correo = String(data[i][15] || "").trim();

  }
  _mapaUsuarios = mapa; return mapa;
}
function usuarioPorCorreo(correo) { return correo ? (mapaUsuarios()[String(correo).trim().toLowerCase()] || null) : null; }

function obtenerDatosControlSaaS(payload) {
  try {
    migrarControlSaaSV2();
    var hoy = hoyIso(payload && payload.hoy);
    var filas = leerFilasControl(getHojaControl());
    var salida = filas.map(function(f) {

// Al cambiar la fecha de suscripción o el tipo se recalcula la fecha de pago; al marcar Cobrado sin fecha de pago se usa hoy.
function completarCambiosControl(fila, cambios, hoy) {
  var r = {}; for (var k in cambios) { r[k] = cambios[k]; }
  var cambiaCiclo = ("fecha_suscripcion" in cambios) || ("tipo_suscripcion" in cambios);
  var inicio = ("fecha_suscripcion" in cambios) ? cambios.fecha_suscripcion : (fila ? fila.fecha_suscripcion : "");
  if (cambiaCiclo && inicio && !("fecha_pago" in cambios)) {
    var tipo = cambios.tipo_suscripcion || (fila ? fila.tipo_suscripcion : "") || "Mensual";
    r.fecha_pago = fechaPagoDeCiclo(inicio, tipo);
  }
  if (cambios.cobrado === true && !(fila && fila.fecha_pago) && !r.fecha_pago) r.fecha_pago = hoy;
  return r;
}

function escribirCambiosFila(hoja, nFila, cambios) {
  for (var campo in cambios) {
    var col = CAMPOS_CONTROL[campo]; if (!col) continue;
    var v = cambios[campo]; var celda = hoja.getRange(nFila, col);
    if (campo === "fecha_suscripcion" || campo === "fecha_pago") { if (v) celda.setValue(parseFechaIso(v)); else celda.clearContent(); }
    else if (campo === "cobrado") celda.setValue(v === true);
    else if (campo === "monto") celda.setValue(Math.max(0, parseFloat(v) || 0));
    else if (campo === "tipo_suscripcion") celda.setValue(v === "Anual" ? "Anual" : "Mensual");
    else celda.setValue(String(v == null ? "" : v).trim());
  }
}

function guardarCambioControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var previo = buscarFilaControl(hoja, String(p.id || ""));
  if (!previo) throw "No se encontró el registro. Recarga la pantalla.";
  var cambios = completarCambiosControl(previo, p.cambios || {}, hoy);
  escribirCambiosFila(hoja, previo.fila, cambios);
  // Como el trigger "reiniciar_avisos": al marcar cobrado o cambiar la fecha de pago se limpian los avisos enviados.
  if (cambios.cobrado === true || ("fecha_pago" in cambios && cambios.fecha_pago !== previo.fecha_pago)) hoja.getRange(previo.fila, 7, 1, 3).clearContent();
  var nueva = buscarFilaControl(hoja, previo.id);
  sincronizarControlConFinanzas(nueva, previo, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Guardado";
}

function insertarFilaControl(hoja, o) {
  var id = Utilities.getUuid(); var fila = primeraFilaLibreControl(hoja);
  hoja.getRange(fila, 1, 1, CONTROL_COLS).setValues([[o.plan || "", o.correo || "", Number(o.monto) || 0, o.cobrado === true, o.fecha_suscripcion ? parseFechaIso(o.fecha_suscripcion) : "", o.fecha_pago ? parseFechaIso(o.fecha_pago) : "", "", "", "", o.comprobante_url || "", "", "", o.metodo_pago || "", o.tipo_suscripcion === "Anual" ? "Anual" : "Mensual", id]]);
  return id;
}

// La fila vacía del final de la tabla: al escribir en ella se crea el registro.
function crearRegistroControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var base = { plan: "", correo: "", monto: 0, cobrado: false, tipo_suscripcion: "Mensual", metodo_pago: "", fecha_suscripcion: "", fecha_pago: "", comprobante_url: "" };
  var cambios = completarCambiosControl(null, p.cambios || {}, hoy);
  for (var k in cambios) { base[k] = cambios[k]; }
  var id = insertarFilaControl(hoja, base);
  sincronizarControlConFinanzas(buscarFilaControl(hoja, id), null, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Registro creado";
}

// Renovar: el nuevo ciclo empieza donde terminó el anterior (fecha de pago en mensual, un año después en anual).
function renovarControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var f = buscarFilaControl(hoja, String(p.id || ""));
  if (!f) throw "No se encontró el registro. Recarga la pantalla.";
  var ciclo = siguienteCiclo(f.fecha_suscripcion, f.fecha_pago, f.tipo_suscripcion, hoy);
  var nuevoMonto = (p.nuevoMonto !== null && p.nuevoMonto !== undefined && p.nuevoMonto !== "") ? Number(p.nuevoMonto) : f.monto;
  var id = insertarFilaControl(hoja, { plan: f.plan, correo: f.correo, monto: nuevoMonto, cobrado: false, metodo_pago: f.metodo_pago, tipo_suscripcion: f.tipo_suscripcion, fecha_suscripcion: ciclo.fecha_suscripcion, fecha_pago: ciclo.fecha_pago });
  sincronizarControlConFinanzas(buscarFilaControl(hoja, id), null, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Se ha generado el nuevo periodo.";
}

function eliminarRegistroControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var f = buscarFilaControl(hoja, String(p.id || ""));
  if (!f) throw "No se encontró el registro. Recarga la pantalla.";
  hoja.deleteRow(f.fila);
  sincronizarControlConFinanzas(null, f, hoy);
  return "Registro eliminado";
}

function filaPagoPorControl(hojaPagos, controlId) {
  var n = hojaPagos.getLastRow(); if (n < 2 || !controlId) return 0;
  var ids = hojaPagos.getRange(2, 10, n - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === controlId) return i + 2; }
  return 0;
}
function insertarPagoDeControl(hojaPagos, f, usuario) {
  var fecha = f.fecha_pago || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var d = parseFechaIso(fecha);
  hojaPagos.appendRow([usuario.uid, usuario.nombre, d, MESES_UP[d.getMonth()], d.getFullYear(), f.tipo_suscripcion, f.metodo_pago || "", f.comprobante_url || "", f.monto, f.id, usuario.correo]);
}

function recalcularSuscripcionSaaS(correo) {
  var usuario = usuarioPorCorreo(correo); if (!usuario) return;
  var hojaEstado = getHojaEstado(); var uid = usuario.uid;
  var filas = leerFilasControl(getHojaControl()).filter(function(f) { return f.correo.toLowerCase() === usuario.correo.toLowerCase(); });
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var tienePagos = false;
  if (np >= 2) { var uids = hojaPagos.getRange(2, 1, np - 1, 1).getValues(); for (var i = 0; i < uids.length; i++) { if (uids[i][0] === uid) { tienePagos = true; break; } } }

  var ne = hojaEstado.getLastRow(); var filaEstado = 0;
  if (ne >= 2) { var ue = hojaEstado.getRange(2, 1, ne - 1, 1).getValues(); for (var j = 0; j < ue.length; j++) { if (ue[j][0] === uid) { filaEstado = j + 2; break; } } }

  if (!filas.length && !tienePagos) { if (filaEstado) hojaEstado.deleteRow(filaEstado); return; }
  var pendiente = null, ultimo = null;
  filas.forEach(function(f) {
    if (!f.fecha_pago) return;
    if (!f.cobrado && (!pendiente || f.fecha_pago < pendiente.fecha_pago)) pendiente = f;
    if (f.cobrado && (!ultimo || f.fecha_pago > ultimo.fecha_pago)) ultimo = f;
  });
  var plan, proximo, base;
  if (pendiente) { plan = pendiente.tipo_suscripcion; proximo = pendiente.fecha_pago; base = pendiente.fecha_suscripcion; }
  else if (ultimo) { plan = ultimo.tipo_suscripcion; proximo = sumarMesesIso(ultimo.fecha_pago, plan === "Anual" ? 12 : 1); base = ultimo.fecha_suscripcion || ultimo.fecha_pago; }
  else return; // solo hay pagos antiguos sin filas de control: se deja como está
  var valores = [[uid, base ? parseFechaIso(base) : "", plan, parseFechaIso(proximo)]];
  if (filaEstado) hojaEstado.getRange(filaEstado, 1, 1, 4).setValues(valores);
  else hojaEstado.getRange(Math.max(2, ne + 1), 1, 1, 4).setValues(valores);
}

function sincronizarControlConFinanzas(nueva, previa, hoy) {
  var hojaPagos = getHojaPagos();
  if (!nueva) { // fila eliminada: el pago se conserva en el historial, solo se desvincula
    var fp = filaPagoPorControl(hojaPagos, previa.id); if (fp) hojaPagos.getRange(fp, 10).setValue("");
    recalcularSuscripcionSaaS(previa.correo); return;
  }
  var usuario = usuarioPorCorreo(nueva.correo);
  if (previa && previa.correo.toLowerCase() !== nueva.correo.toLowerCase()) {
    var fm = filaPagoPorControl(hojaPagos, nueva.id);
    if (fm && usuario) { hojaPagos.getRange(fm, 1, 1, 2).setValues([[usuario.uid, usuario.nombre]]); hojaPagos.getRange(fm, 11).setValue(usuario.correo); }
    if (previa.correo) recalcularSuscripcionSaaS(previa.correo);
  }
  if (nueva.cobrado && usuario) {
    var fila = filaPagoPorControl(hojaPagos, nueva.id);
    if (!fila) insertarPagoDeControl(hojaPagos, nueva, usuario);
    else if (previa && previa.cobrado && (previa.monto !== nueva.monto || previa.comprobante_url !== nueva.comprobante_url || previa.metodo_pago !== nueva.metodo_pago)) {
      var actual = hojaPagos.getRange(fila, 7, 1, 3).getValues()[0]; // G método · H recibo · I monto
      hojaPagos.getRange(fila, 7, 1, 3).setValues([[nueva.metodo_pago || actual[0], nueva.comprobante_url || actual[1], nueva.monto]]);
    }
  } else if (previa && previa.cobrado && !nueva.cobrado) {
    var fe = filaPagoPorControl(hojaPagos, nueva.id); if (fe) hojaPagos.deleteRow(fe);
  }
  recalcularSuscripcionSaaS(nueva.correo);
}

// Una sola vez: IDs, tipo "Mensual" por defecto y pago en el historial de cada cuota ya cobrada.
function migrarControlSaaSV2() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty("CONTROL_SAAS_V2") === "ok") return;
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);
  if (filas.length) {
    var primera = filas[0].fila; var ultima = filas[filas.length - 1].fila;
    var rango = hoja.getRange(primera, 14, ultima - primera + 1, 1); var vals = rango.getValues();
    filas.forEach(function(f) { if (vals[f.fila - primera][0] === "") vals[f.fila - primera][0] = "Mensual"; });
    rango.setValues(vals);
  }
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var pagos = np >= 2 ? hojaPagos.getRange(2, 1, np - 1, 11).getValues() : [];
  filas.forEach(function(f) {
    var usuario = usuarioPorCorreo(f.correo); if (!f.cobrado || !usuario) return;
    if (filaPagoPorControl(hojaPagos, f.id)) return;
    var d = parseFechaIso(f.fecha_pago); var vinculado = 0;
    for (var i = 0; i < pagos.length; i++) { // pago manual antiguo del mismo usuario, mes y monto: se vincula en vez de duplicarlo
      if (pagos[i][0] === usuario.uid && !pagos[i][9] && d && String(pagos[i][3]).toUpperCase() === MESES_UP[d.getMonth()] && Number(pagos[i][4]) === d.getFullYear() && Number(pagos[i][8]) === f.monto) { vinculado = i + 2; pagos[i][9] = f.id; break; }
    }
    if (vinculado) hojaPagos.getRange(vinculado, 10, 1, 2).setValues([[f.id, usuario.correo]]);
    else insertarPagoDeControl(hojaPagos, f, usuario);
  });
  var vistos = {};
  filas.forEach(function(f) { var k = f.correo.toLowerCase(); if (f.correo && !vistos[k]) { vistos[k] = true; recalcularSuscripcionSaaS(f.correo); } });
  props.setProperty("CONTROL_SAAS_V2", "ok");
}

function rellenarPlantillaCobro(plantilla, d) {
  return String(plantilla).split("[Nombre]").join(d.nombre).split("[Proyecto]").join(d.proyecto).split("[Monto]").join(String(d.monto));
}
function escaparHtml(s) { return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;"); }

// Asunto sobrio (sin emojis) + texto y HTML + pie que identifica quién envía: toleran mejor los filtros anti-spam.
function construirCorreoCobro(d) {
  var concepto = String(d.concepto || "Pendiente").replace(/[\r\n]+/g, " ").trim().substring(0, 80);
  var firma = d.remitente || d.responderA;
  var pie = "Recibes este mensaje de " + firma + " por un pago pendiente. Si ya realizaste el pago, ignora este recordatorio o responde a este correo.";
  var cuerpo = String(d.cuerpo).trim();
  var text = cuerpo + "\n\nSaludos,\n" + firma + "\n\n--\n" + pie;
  var parrafos = cuerpo.split(/\n{2,}/).map(function(p) { return '<p style="margin:0 0 14px">' + escaparHtml(p).replace(/\n/g, "<br>") + "</p>"; }).join("");
  var html = '<div style="font-family:Arial,Helvetica,sans-serif;font-size:15px;line-height:1.5;color:#222;max-width:560px">' + parrafos +
    '<p style="margin:18px 0 0">Saludos,<br>' + escaparHtml(firma) + "</p>" +
    '<hr style="border:none;border-top:1px solid #ddd;margin:22px 0 10px">' +
    '<p style="margin:0;font-size:12px;color:#777">' + escaparHtml(pie) + "</p></div>";
  return { subject: "Recordatorio de pago de tu plan: " + concepto, text: text, html: html };
}

function obtenerConfigRemitente() {
  var props = PropertiesService.getScriptProperties();
  var alias = props.getProperty("COBRO_ALIAS") || "";
  return { alias: alias, nombre: props.getProperty("COBRO_NOMBRE") || "", direccion: alias || Session.getEffectiveUser().getEmail() };
}

function obtenerRemitenteCobro() {
  var c = obtenerConfigRemitente(); var aliases = [];
  try { aliases = GmailApp.getAliases(); } catch (e) {}
  return { direccion: Session.getEffectiveUser().getEmail(), alias: c.alias, nombre: c.nombre, aliases: aliases };
}

function guardarRemitenteCobro(p) {
  var alias = String(p.alias || "").trim(); var nombre = String(p.nombre || "").replace(/["<>\r\n]/g, "").trim();
  if (alias) {
    var aliases = []; try { aliases = GmailApp.getAliases(); } catch (e) {}
    if (aliases.indexOf(alias) === -1) throw "Ese correo no está configurado como «Enviar como» en tu Gmail.";
  }
  var props = PropertiesService.getScriptProperties();
  props.setProperty("COBRO_ALIAS", alias); props.setProperty("COBRO_NOMBRE", nombre);
  return "Tus recordatorios ahora saldrán desde " + (alias || Session.getEffectiveUser().getEmail()) + ".";
}

function ejecutarEnvioCobroSaaS(p) {
  var hoy = hoyIso(p && p.hoy);
  var hojaScripts = obtenerHojaScripts();
  var scripts = [hojaScripts.getRange("B6").getValue(), hojaScripts.getRange("B14").getValue(), hojaScripts.getRange("B22").getValue()];
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja); var pendientes = [];
  filas.forEach(function(f) { var et = etapaPendienteFila(f, hoy); if (et >= 0) pendientes.push({ f: f, etapa: et }); });
  if (!pendientes.length) { actualizarEstadosVencimientoSaaS(hoy); return { mensaje: "No hay recordatorios pendientes por enviar.", enviados: 0 }; }

  var rem = obtenerConfigRemitente(); var mapa = mapaUsuarios();
  var cuota = MailApp.getRemainingDailyQuota();
  var enviados = 0, sinCorreo = 0, sinPlantilla = 0, sinCuota = 0; var fallidos = []; var primero = true;

  for (var i = 0; i < pendientes.length; i++) {
    var f = pendientes[i].f; var etapa = pendientes[i].etapa;
    if (!f.correo || f.correo.indexOf("@") < 0) { sinCorreo++; continue; }
    if (!scripts[etapa]) { sinPlantilla++; continue; }
    if (enviados >= cuota) { sinCuota++; continue; }
    var u = mapa[f.correo.toLowerCase()];
    var nombre = (u && u.nombre) ? u.nombre : f.correo.split("@")[0];
    var texto = rellenarPlantillaCobro(scripts[etapa], { nombre: nombre, proyecto: f.plan || "Suscripción", monto: f.monto });
    var correo = construirCorreoCobro({ concepto: f.plan || "Suscripción", cuerpo: texto, remitente: rem.nombre, responderA: rem.direccion });
    if (!primero) Utilities.sleep(800); // una ráfaga idéntica se parece a spam masivo
    primero = false;
    try {
      var op = { htmlBody: correo.html, replyTo: rem.direccion };
      if (rem.nombre) op.name = rem.nombre;
      if (rem.alias) op.from = rem.alias;
      GmailApp.sendEmail(f.correo, correo.subject, correo.text, op);
      enviados++;
      hoja.getRange(f.fila, ETAPAS_COBRO[etapa].col).setValue("ENVIADO");
    } catch (e) { fallidos.push(f.correo + " (" + String(e.message || e).split("\n")[0].substring(0, 200) + ")"); }
  }
  actualizarEstadosVencimientoSaaS(hoy);
  var mensaje = "Se enviaron " + enviados + " recordatorios de cobro.";
  if (sinCorreo) mensaje += " " + sinCorreo + " vencidos no tienen un correo válido.";
  if (sinPlantilla) mensaje += " " + sinPlantilla + " no se enviaron porque su plantilla está vacía.";
  if (sinCuota) mensaje += " " + sinCuota + " quedaron pendientes por el límite diario de correos de Google.";
  if (fallidos.length) mensaje += " Fallaron: " + fallidos.join("; ") + ".";
  return { mensaje: mensaje, enviados: enviados };
}

// REPORTE FINANCIERO ADMIN (cobros con fecha de pago; cliente = correo del usuario)
// ==========================================
function obtenerDatosReporteAnualSaaS(anio) {
  try {
    var meses = []; for (var m = 0; m < 12; m++) meses.push(0); var total = 0;
    leerFilasControl(getHojaControl()).forEach(function(f) {
      if (!f.cobrado || !f.fecha_pago) return;
      if (Number(f.fecha_pago.substring(0, 4)) === Number(anio)) { meses[Number(f.fecha_pago.substring(5, 7)) - 1] += f.monto; total += f.monto; }
    });
    return { meses: meses, total: total };
  } catch (e) { return { error: e.toString() }; }
}

function obtenerDatosReporteClientesSaaS(anio, mesTexto) {
  try {
    var mesIndex = MESES_UP.indexOf(String(mesTexto).toUpperCase()); var mapa = {}; var orden = [];
    leerFilasControl(getHojaControl()).forEach(function(f) {
      if (!f.cobrado || !f.fecha_pago) return;
      if (Number(f.fecha_pago.substring(0, 4)) === Number(anio) && Number(f.fecha_pago.substring(5, 7)) - 1 === mesIndex) {
        var cliente = f.correo || "Sin Usuario";
        if (mapa[cliente] === undefined) { mapa[cliente] = 0; orden.push(cliente); }
        mapa[cliente] += f.monto;
      }
    });
    return { labels: orden, valores: orden.map(function(c) { return mapa[c]; }) };
  } catch (e) { return { error: e.toString() }; }
}
