// AUTENTICACIÓN REAL: se valida el ID token de Firebase en el servidor.
// El uid y el correo salen del token, NUNCA de lo que mande el navegador.
// ==========================================
function verificarToken(idToken) {
  if (!idToken || typeof idToken !== "string" || idToken.length < 100 || idToken.length > 4000) throw "Sesión inválida";
  var cache = CacheService.getScriptCache();
  var key = "tok_" + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, idToken));
  var hit = cache.get(key);
  if (hit) return JSON.parse(hit);
  var resp = UrlFetchApp.fetch("https://identitytoolkit.googleapis.com/v1/accounts:lookup?key=" + FIREBASE_API_KEY, {
    method: "post", contentType: "application/json", payload: JSON.stringify({ idToken: idToken }), muteHttpExceptions: true
  });
  if (resp.getResponseCode() !== 200) throw "Sesión inválida";
  var users = JSON.parse(resp.getContentText()).users;
  if (!users || !users.length || users[0].disabled) throw "Sesión inválida";
  var u = users[0];
  var out = { uid: String(u.localId), email: String(u.email || ""), verified: (u.emailVerified === true) };
  cache.put(key, JSON.stringify(out), 300);
  return out;
}

  if (typeof e === "string") return e;

  return "Error interno del servidor. Intenta de nuevo.";
}

function buscarFilaUsuario(dbSheet, userId) {

function limpiarTelefono(t) {
  var s = String(t == null ? "" : t).trim();
  if (!s) return "";
  if (!/^\+?[\d\s\-()]{7,25}$/.test(s)) throw "Teléfono inválido. Usa solo números, espacios y el signo + (ej: +584121234567).";
  return s;
}
function escribirTelefono(dbSheet, fila, tel) {
  var c = dbSheet.getRange(fila, 18);
  c.setNumberFormat("@");   // texto: conserva el "+" y los ceros iniciales
  c.setValue(tel);
}

// Crea la fila del usuario si no existe (estado Pendiente) y completa el correo si falta. Devuelve el número de fila.
function asegurarFilaUsuario(dbSheet, userId, email, verificado) {

  if (fila) {
    if (correo && String(dbSheet.getRange(fila, 16).getValue() || "") === "") dbSheet.getRange(fila, 16).setValue(correo);
    return fila;
  }
  // primera fila libre según la columna A (la celda N2 del ícono no cuenta)
  var n = Math.max(2, dbSheet.getLastRow());
  var colA = dbSheet.getRange(2, 1, n - 1, 1).getValues();
  var ultima = 1;
  for (var i = 0; i < colA.length; i++) { if (colA[i][0] !== "") ultima = i + 2; }

function parseFloatSafe(val) { if (val === "" || val === undefined || val === null) return ""; var str = String(val).replace(',', '.'); var num = parseFloat(str); return isNaN(num) ? "" : num; }
function readImageUrl(sheet, cellStr) { var formula = sheet.getRange(cellStr).getFormula(); if (formula && formula.toUpperCase().indexOf('=IMAGE') !== -1) { var match = formula.match(/=IMAGE\("([^"]+)"/i); if (match) return match[1]; } return sheet.getRange(cellStr).getDisplayValue(); }
function writeImageUrl(sheet, cellStr, url) {
  var range = sheet.getRange(cellStr);
  var u = String(url == null ? "" : url).trim();
  if (!u) { range.clearContent(); return; }
  if (!/^https?:\/\//i.test(u) || u.length > 2000) throw "El enlace de la imagen no es válido (debe iniciar con http:// o https://).";
  var directUrl = u;
  var match = u.match(/drive\.google\.com\/(?:file\/d\/|open\?id=)([-_a-zA-Z0-9]+)/);
  if (match && match[1]) { directUrl = "https://drive.google.com/thumbnail?id=" + match[1] + "&sz=w1000"; }
  // Se codifican comillas, espacios y barras: la URL no puede "salirse" de la fórmula =IMAGE("...")
  directUrl = directUrl.replace(/"/g, "%22").replace(/\s/g, "%20").replace(/\\/g, "%5C");
  range.setFormula('=IMAGE("' + directUrl + '")');
}

// ---- Saneamiento de entradas (NUEVO) ----
// Evita la inyección de fórmulas: un texto que empieza por = + - @ se guarda como texto literal.
function celdaSegura(v) { v = (v == null) ? "" : String(v); return /^[=+\-@]/.test(v) ? "'" + v : v; }
function textoSeguro(v, max) {
  var s = (v == null) ? "" : String(v);
  s = s.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, "").trim();
  if (max && s.length > max) s = s.substring(0, max);
  return celdaSegura(s);
}
function round2(x) { return Math.round(x * 100) / 100; }

// Acepta DD/MM/AAAA o AAAA-MM-DD y devuelve un Date real (sin ambigüedad de idioma de la hoja) o "".
function parseFechaEntrada(v) {
  if (v instanceof Date) return isNaN(v.getTime()) ? "" : v;
  var s = String(v == null ? "" : v).trim();
  if (!s) return "";
  var m = s.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})$/), d, mo, y;
  if (m) { d = +m[1]; mo = +m[2]; y = +m[3]; }
  else { m = s.match(/^(\d{4})-(\d{2})-(\d{2})/); if (!m) return ""; y = +m[1]; mo = +m[2]; d = +m[3]; }
  var f = new Date(y, mo - 1, d);
  if (f.getFullYear() !== y || f.getMonth() !== mo - 1 || f.getDate() !== d) return "";
  if (y < 1900 || y > 2100) return "";
  return f;
}

function sumarMesesFecha(d, n) {
  var y = d.getFullYear(), m = d.getMonth() + n, dia = d.getDate();
  var ultimo = new Date(y, m + 1, 0).getDate();
  return new Date(y, m, Math.min(dia, ultimo));   // 31-ene + 1 mes = 28/29-feb (no desborda a marzo)
}

function calcularEdadExacta(fNac, hoy) {
  var edad = hoy.getFullYear() - fNac.getFullYear();
  var m = hoy.getMonth() - fNac.getMonth();
  if (m < 0 || (m === 0 && hoy.getDate() < fNac.getDate())) edad--;
  return edad;
}

// Talla en metros. Si el usuario escribe centímetros (ej. 175) se convierte; fuera de rango = inválida.
function normalizarTalla(v) {
  var t = parseFloatSafe(v);
  if (t === "") return "";
  if (t > 3 && t < 260) t = t / 100;
  if (t < 0.5 || t > 2.6) return "";
  return Math.round(t * 100) / 100;
}

  var g = calcSheet ? String(calcSheet.getRange("D2").getDisplayValue() || "").trim() : "";
  return g || "Hombre";
}
// Mifflin-St Jeor: +5 hombres, -161 mujeres (antes la fila mensual usaba siempre +5)
function constanteTMB(genero) { return (genero && String(genero).toLowerCase().indexOf("mujer") !== -1) ? -161 : 5; }
function calcularDeficitCalorico(peso, tallaM, edad, genero) {
  var tmb = (10 * peso) + (6.25 * (tallaM * 100)) - (5 * edad) + constanteTMB(genero);
  return Math.round((tmb * 1.2) - 500);
}
function esFilaBalance(r) {

  for (i = 0; i < JROWS_BALANCE.length; i++) { if (JROWS_BALANCE[i].r === r) return true; }

}

// Calcula diferencia exacta emulando SIFECHA de Excel
function calcularDiferenciaMesesDias(fechaInicio, fechaFin) {
  if (!fechaInicio || !fechaFin) return "";
  var d1 = new Date(fechaInicio);
  var d2 = new Date(fechaFin);
  if (isNaN(d1.getTime()) || isNaN(d2.getTime())) return "";

  var m1 = d1.getFullYear() * 12 + d1.getMonth();
  var m2 = d2.getFullYear() * 12 + d2.getMonth();
  var meses = m2 - m1;
  var dias = d2.getDate() - d1.getDate();

  if (dias < 0) {
    meses--;
    var ultimoDiaMesAnterior = new Date(d2.getFullYear(), d2.getMonth(), 0).getDate();
    dias += ultimoDiaMesAnterior;
  }

  if (meses < 0) return "";
  return meses + " meses, " + dias + " días";
}

// forzarFechas = true recalcula las fechas de los meses 2..13 desde la fecha de registro (C2);
// en false solo rellena las vacías, respetando las fechas de medición que escribió el usuario.
function restaurarFormulasPerfil(sheet, forzarFechas) {
  var ss = sheet.getParent();
  var calcSheet = ss.getSheetByName("Calculadora") || ss.getSheetByName("Calculadoras");
  var diarioSheet = ss.getSheetByName("Diario");
  var today = new Date(); today.setHours(0, 0, 0, 0);
  var genero = obtenerGenero(calcSheet);

  // Edad exacta (antes se dividía entre 365.25 días y podía fallar un día)
  var fNacVal = sheet.getRange("A19").getValue();
  var edadCalc = "";
  if (fNacVal instanceof Date && !isNaN(fNacVal.getTime())) { var ed = calcularEdadExacta(fNacVal, today); if (ed >= 0 && ed <= 120) edadCalc = ed; }
  sheet.getRange("A4").setValue(edadCalc);
  var edadVal = edadCalc;
  var tallaVal = normalizarTalla(sheet.getRange("A7").getValue());
  var pesoObj = parseFloatSafe(sheet.getRange("A16").getValue());
  if (calcSheet) { sheet.getRange("A13").setValue(calcSheet.getRange("D6").getValue()); }

  // Fechas de los meses
  var c2Val = sheet.getRange("C2").getValue();
  sheet.getRange("A31").setValue(c2Val);
  var c2Date = (c2Val instanceof Date && !isNaN(c2Val.getTime())) ? c2Val : null;
  var filaFechas = sheet.getRange("C2:O2").getValues()[0];
  for (var i = 1; i < filaFechas.length; i++) {
    if (!c2Date) { if (forzarFechas) filaFechas[i] = ""; continue; }
    if (forzarFechas || filaFechas[i] === "") filaFechas[i] = sumarMesesFecha(c2Date, i);
  }
  sheet.getRange("C2:O2").setValues([filaFechas]);

  // Peso, diferencia con el objetivo y déficit calórico de cada mes
  sheet.getRange("C3").setValue(sheet.getRange("A10").getValue());
  var pesosFila = sheet.getRange("C3:O3").getValues()[0];
  var f4 = [], f19 = [];
  for (var c = 0; c < pesosFila.length; c++) {
    var pm = parseFloatSafe(pesosFila[c]);
    f4.push((pm !== "" && pesoObj !== "") ? round2(pm - pesoObj) : "");
    f19.push((pm !== "" && tallaVal !== "" && edadVal !== "") ? calcularDeficitCalorico(pm, tallaVal, edadVal, genero) : "");
  }
  sheet.getRange("C4:O4").setValues([f4]);
  sheet.getRange("C19:O19").setValues([f19]);

  // Escalones
  var filasA = [42, 53, 64, 75, 86, 97, 108, 119, 130, 141, 152, 163];
  for (var k = 0; k < filasA.length; k++) {
    sheet.getRange("A" + filasA[k]).setValue(filaFechas[k + 1]);   // D2..O2
    var kVal = sheet.getRange("K" + filasA[k]).getValue();
    sheet.getRange("L" + filasA[k]).setValue((c2Date && kVal instanceof Date) ? calcularDiferenciaMesesDias(c2Date, kVal) : "");
  }

  sheet.getRange("K23").setValue(today);
  if (diarioSheet) {
    var pesoActualDiario = diarioSheet.getRange("B382").getValue();
    sheet.getRange("K27").setValue(diarioSheet.getRange("C382").getValue());
    sheet.getRange("K31").setValue(pesoActualDiario);
    if (calcSheet) {
      calcSheet.getRange("D4").setValue(pesoActualDiario);
      var pNum = parseFloatSafe(pesoActualDiario);
      if (pNum !== "" && tallaVal !== "") { calcSheet.getRange("D7").setValue(Math.round((pNum / (tallaVal * tallaVal)) * 10) / 10); }
      else { calcSheet.getRange("D7").setValue(""); }
    }
  }
}

function restaurarFormulasDiario(sheet) {
  var ss = sheet.getParent();
  var perfilSheet = ss.getSheetByName("Perfil");
  var calcSheet = ss.getSheetByName("Calculadora") || ss.getSheetByName("Calculadoras");
  var baseRaw = perfilSheet.getRange("C2").getValue();
  var baseDate = (baseRaw instanceof Date && !isNaN(baseRaw.getTime())) ? baseRaw : parseFechaEntrada(baseRaw);
  var pesoInicial = parseFloatSafe(perfilSheet.getRange("C3").getValue());
  var edadVal = parseFloatSafe(perfilSheet.getRange("A4").getValue());
  var edad = (edadVal !== "") ? edadVal : 30;   // sin fecha de nacimiento se asume 30 (el dashboard avisa con "edadAsumida")
  var genero = obtenerGenero(calcSheet);

  function calcBP(bpStr, ed) {
    if (!bpStr || bpStr === "") return "";
    var parts = String(bpStr).replace(/\s/g, "").split("/");
    if (parts.length !== 2) return "Formato incorrecto";
    var sys = parseInt(parts[0], 10), dia = parseInt(parts[1], 10);
    if (isNaN(sys) || isNaN(dia)) return "Formato incorrecto";
    var t2_sys = (ed < 50) ? 140 : (ed < 70 ? 150 : 160);
    var t1_sys = (ed < 50) ? 130 : (ed < 70 ? 140 : 150);
    var te_sys = 120 + (Math.max(3, Math.min(8, Math.floor(ed / 10))) - 3) * 5;
    if (sys < 90 || dia < 60) return "Baja";
    if (sys >= t2_sys || dia >= 90) return "Etapa 2 Alta";
    if (sys >= t1_sys || dia >= 80) return "Alta";
    if (sys >= te_sys && dia < 80) return "Elevada";
    return "Normal";
  }

  var dataFull = sheet.getRange("B2:H381").getValues();
  var dates = [], diffsC = [], diffsD = [], bpResults = [];
  var sumC = 0, lastPeso = "", ultimoPeso = "", lastF = "--", lastG = "--", lastH = "--";

  for (var i = 0; i < 380; i++) {
    dates.push([baseDate ? new Date(baseDate.getFullYear(), baseDate.getMonth(), baseDate.getDate() + i) : ""]);
    var pCur = parseFloatSafe(dataFull[i][0]);

    // Diferencia contra el ÚLTIMO peso registrado (antes, si faltaba el día anterior se restaba contra 0 y salía "+82 kg")
    if (i > 0) {
      if (pCur !== "" && ultimoPeso !== "") { var diff = round2(pCur - ultimoPeso); diffsC.push([diff]); sumC += diff; }
      else { diffsC.push([""]); }
    }
    if (pCur !== "") { lastPeso = pCur; ultimoPeso = pCur; }

    // Diferencia contra el peso inicial (si no hay peso inicial queda vacía, ya no se resta 0)
    diffsD.push([(pCur !== "" && pesoInicial !== "") ? round2(pCur - pesoInicial) : ""]);

    if (dataFull[i][4] !== "") lastF = dataFull[i][4];
    if (dataFull[i][5] !== "") lastG = dataFull[i][5];
    if (dataFull[i][6] !== "") lastH = dataFull[i][6];
    bpResults.push([calcBP(dataFull[i][6], edad)]);
  }
  sheet.getRange("A2:A381").setValues(dates);
  sheet.getRange("C3:C381").setValues(diffsC);
  sheet.getRange("D2:D381").setValues(diffsD);
  sheet.getRange("I2:I381").setValues(bpResults);
  sheet.getRange("B382").setValue(lastPeso);
  sheet.getRange("C382").setValue(round2(sumC));
  sheet.getRange("F382").setValue(lastF);
  sheet.getRange("G382").setValue(lastG);
  sheet.getRange("H382").setValue(lastH);
  sheet.getRange("I382").setValue(calcBP(lastH !== "--" ? lastH : "", edad));

  // Balances (columna J). Solo se escribe lo que cambió.
  var jActual = sheet.getRange("J2:J381").getValues();
  function setJ(r, val) { if (jActual[r - 2][0] !== val) sheet.getRange("J" + r).setValue(val); }

  // Balances Semanales
  for (var j = 0; j < JROWS_BALANCE.length; j++) {
    var map = JROWS_BALANCE[j];
    var v1 = parseFloatSafe(dataFull[map.b1 - 2][0]); var v2 = parseFloatSafe(dataFull[map.b2 - 2][0]);
    setJ(map.r, (v1 !== "" && v2 !== "") ? round2(v2 - v1) : "");
  }
  // Balances Mensuales
  for (var m = 0; m < MROWS_BALANCE.length; m++) {
    var mapM = MROWS_BALANCE[m];
    var vs = parseFloatSafe(dataFull[mapM.startRow - 2][0]); var ve = parseFloatSafe(dataFull[mapM.endRow - 2][0]);
    setJ(mapM.r, (vs !== "" && ve !== "") ? "(BALANCE MENSUAL) = " + (ve - vs).toFixed(2) + " Kg" : "");
  }

  // Déficit calórico diario en base al peso actual (misma fórmula que las filas mensuales, con género)
  var deficitCalculado = "";
  if (lastPeso !== "") {
    var tallaVal = normalizarTalla(perfilSheet.getRange("A7").getValue());
    if (tallaVal !== "") deficitCalculado = calcularDeficitCalorico(lastPeso, tallaVal, edad, genero);
  }
  sheet.getRange("K382").setValue(deficitCalculado);
}

function saveWelcomeData(datos, userId) {
  var userEnv = getUserSpreadsheetAutorizado(userId);
  var sheet = userEnv.ss.getSheetByName("Perfil");
  var calcSheet = userEnv.ss.getSheetByName("Calculadora") || userEnv.ss.getSheetByName("Calculadoras");
  var diarioSheet = userEnv.ss.getSheetByName("Diario");

  var nombre = textoSeguro(datos.nombre, 60);
  if (nombre) sheet.getRange("A2").setValue(nombre);

  // Fecha de registro (si cambia se recalculan las fechas de los meses)
  var cambioFechaReg = false;
  if (datos.fechaReg) {
    var fr = parseFechaEntrada(datos.fechaReg);
    if (fr === "") throw "Fecha de registro inválida. Usa DD/MM/AAAA.";
    var c2Actual = sheet.getRange("C2").getValue();
    cambioFechaReg = !(c2Actual instanceof Date) || isoDesdeFecha(c2Actual) !== isoDesdeFecha(fr);
    sheet.getRange("C2").setValue(fr);
  }
  if (datos.fecha) {
    var fn = parseFechaEntrada(datos.fecha);
    if (fn === "") throw "Fecha de nacimiento inválida. Usa DD/MM/AAAA.";
    if (fn.getTime() > new Date().getTime()) throw "La fecha de nacimiento no puede ser futura.";
    sheet.getRange("A19").setValue(fn);
  }
  if (datos.pesoIni) {
    var pi = parseFloatSafe(datos.pesoIni);
    if (pi === "" || pi < 20 || pi > 400) throw "Peso inicial inválido (entre 20 y 400 kg).";
    sheet.getRange("A10").setValue(pi);
  }
  var tallaNum = "";
  if (datos.talla !== undefined && String(datos.talla).trim() !== "") {
    tallaNum = normalizarTalla(datos.talla);
    if (tallaNum === "") throw "Talla inválida. Escríbela en metros (ej: 1.75).";
    sheet.getRange("A7").setValue(tallaNum);
    if (calcSheet) calcSheet.getRange("D3").setValue(tallaNum);
  }
  if (datos.pesoObj !== undefined) {
    var po = parseFloatSafe(datos.pesoObj);
    if (po !== "" && (po < 20 || po > 400)) throw "Peso objetivo inválido (entre 20 y 400 kg).";
    sheet.getRange("A16").setValue(po);
  }
  if (datos.avatarUrl !== undefined) writeImageUrl(sheet, "A21", datos.avatarUrl);

  // Género: ya no se asume "Hombre" cuando no llega; se conserva el guardado
  var generoUser = obtenerGenero(calcSheet);
  if (datos.genero === "Hombre" || datos.genero === "Mujer") {
    generoUser = datos.genero;
    if (calcSheet) calcSheet.getRange("D2").setValue(generoUser);
  }

  // Peso ideal (Lorentz) con la talla guardada y el género actual
  var tallaRef = (tallaNum !== "") ? tallaNum : normalizarTalla(sheet.getRange("A7").getValue());
  if (tallaRef !== "" && calcSheet) {
    var tallaCm = tallaRef * 100;
    var pesoIdealCalc = (generoUser === "Mujer") ? (tallaCm - 100 - ((tallaCm - 150) / 2.5)) : (tallaCm - 100 - ((tallaCm - 150) / 4));
    var pIdArr = Math.round(pesoIdealCalc * 10) / 10;
    calcSheet.getRange("D6").setValue(pIdArr);
    sheet.getRange("A13").setValue(pIdArr);
  }

  restaurarFormulasPerfil(sheet, cambioFechaReg);
  if (diarioSheet) { restaurarFormulasDiario(diarioSheet); restaurarFormulasPerfil(sheet, false); }

function saveMedidasMesUniversal(payload, userId) {
  var userEnv = getUserSpreadsheetAutorizado(userId);
  var sheet = userEnv.ss.getSheetByName("Perfil");
  var mes = parseInt(payload.mes, 10);
  if (!(mes >= 1 && mes <= 12)) throw "Mes inválido.";
  var colLetra = String.fromCharCode(66 + mes);   // mes 1 = C ... mes 12 = N

  // Fecha: el mes 1 es la fecha de registro (C2); en los demás, si se deja vacía se autocalcula
  var cambioFechaReg = false;
  var fecha = parseFechaEntrada(payload.fechaMed);
  if (payload.fechaMed && fecha === "") throw "Fecha inválida. Usa DD/MM/AAAA.";
  if (mes === 1) {
    if (fecha !== "") {
      var c2Actual = sheet.getRange("C2").getValue();
      cambioFechaReg = !(c2Actual instanceof Date) || isoDesdeFecha(c2Actual) !== isoDesdeFecha(fecha);
      sheet.getRange("C2").setValue(fecha);
    }
  } else {
    sheet.getRange(colLetra + "2").setValue(fecha);
    // El peso del mes 1 es siempre el peso inicial (A10) y la fila 4 se calcula sola: solo se guarda el peso de los meses 2-12
    var pesoMes = parseFloatSafe(payload.pesoColB);
    if (pesoMes !== "" && (pesoMes < 20 || pesoMes > 400)) throw "Peso inválido (entre 20 y 400 kg).";
    sheet.getRange(colLetra + "3").setValue(pesoMes);
  }

  // 14 medidas, filas 5..18, en el MISMO orden en que el dashboard las lee (antes faltaban brazo derecho y pantorrilla derecha y todo se corría una fila)
  var campos = ["cuello", "hombro", "pecho", "cintura", "brazoI", "brazoD", "anteI", "anteD", "circAb", "musloI", "musloD", "pantI", "pantD", "cadera"];
  var vals = campos.map(function(c) {
    var v = parseFloatSafe(payload[c]);
    if (v !== "" && (v <= 0 || v > 300)) throw "Medida inválida en: " + c;
    return [v];
  });
  sheet.getRange(colLetra + "5:" + colLetra + "18").setValues(vals);

  var fM = 23 + ((mes - 1) * 11);
  writeImageUrl(sheet, "B" + fM, payload.fotoFrente);
  writeImageUrl(sheet, "D" + fM, payload.fotoEspalda);

  CacheService.getScriptCache().remove("dash_" + userId);
  return "Medidas actualizadas";
}

// ---------- Fechas (mismas reglas que lib/fechas.ts de Freelancer) ----------
function aIsoFecha(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd");
  if (typeof v === "string") { var m = v.match(/^(\d{4})-(\d{2})-(\d{2})/); if (m) return m[1] + "-" + m[2] + "-" + m[3]; }
  return "";
}
function parseFechaIso(iso) { var m = String(iso || "").match(/^(\d{4})-(\d{2})-(\d{2})/); return m ? new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])) : null; }
function isoDesdeFecha(d) { var mm = ("0" + (d.getMonth() + 1)).slice(-2); var dd = ("0" + d.getDate()).slice(-2); return d.getFullYear() + "-" + mm + "-" + dd; }
// La fecha del navegador solo se acepta si está a ±1 día de la del servidor (evita fechas manipuladas)
function hoyIso(hoyCliente) {
  var servidor = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  if (/^\d{4}-\d{2}-\d{2}$/.test(hoyCliente || "")) { var dd = diasDesdeIso(hoyCliente, servidor); if (dd !== null && Math.abs(dd) <= 1) return hoyCliente; }
  return servidor;
}

function diasDesdeIso(iso, hoy) { var f = parseFechaIso(iso), r = parseFechaIso(hoy); if (!f || !r) return null; return Math.round((r.getTime() - f.getTime()) / 86400000); }
// 30-ago + 1 mes = 30-sep y 31-ago + 1 mes = 30-sep (no desborda al mes siguiente)
function sumarMesesIso(iso, meses) {
  var f = parseFechaIso(iso) || new Date(); var dia = f.getDate();
  f.setDate(1); f.setMonth(f.getMonth() + meses);
  var ultimo = new Date(f.getFullYear(), f.getMonth() + 1, 0).getDate();
  f.setDate(Math.min(dia, ultimo));
  return isoDesdeFecha(f);
}
// Mensual: mismo día del mes siguiente · Anual: la misma fecha
function fechaPagoDeCiclo(fsusc, tipo) { return tipo === "Anual" ? fsusc : sumarMesesIso(fsusc, 1); }
function siguienteCiclo(fsusc, fpago, tipo, hoy) {
  var inicio = (tipo === "Anual") ? sumarMesesIso(fsusc || fpago || hoy, 12) : (fpago || (fsusc ? sumarMesesIso(fsusc, 1) : hoy));
  return { fecha_suscripcion: inicio, fecha_pago: fechaPagoDeCiclo(inicio, tipo) };
}

  _filasCache = filas;
  return filas;
}

function buscarFilaControl(hoja, id) {
  var filas = leerFilasControl(hoja);
  for (var i = 0; i < filas.length; i++) { if (filas[i].id === id) return filas[i]; }
  return null;
}

function primeraFilaLibreControl(hoja) {
  var ultima = hoja.getLastRow();
  if (ultima < CONTROL_PRIMERA_FILA) return CONTROL_PRIMERA_FILA;
  var vals = hoja.getRange(CONTROL_PRIMERA_FILA, 1, ultima - CONTROL_PRIMERA_FILA + 1, CONTROL_COLS).getValues();
  for (var i = 0; i < vals.length; i++) {
    var f = vals[i];
    if (f[0] === "" && f[1] === "" && f[2] === "" && f[4] === "" && f[5] === "" && f[9] === "" && f[12] === "" && f[14] === "") return CONTROL_PRIMERA_FILA + i;
  }
  return ultima + 1;
}

// Vence Hoy / 7 Días / 15 Días: "ENVIADO" si ya se mandó, "VENCIDO…" si ya pasó el plazo, "" si nada.
function marcaHecha(m) { return m === "ENVIADO" || m === "OMITIDO"; }
function estadosVencimientoFila(f, hoy) {
  var dias = f.cobrado ? null : diasDesdeIso(f.fecha_pago, hoy);
  return ETAPAS_COBRO.map(function(et, i) {
    if (marcaHecha(f.marcas[i])) return f.marcas[i];
    if (dias !== null && dias >= et.dias) return et.etiqueta;
    return "";
  });
}

// Se envía una sola etapa por ejecución y en orden: hoy → 7D → 15D.
// Se envía SOLO la etapa más alta que ya corresponde (antes, con 20 días de atraso se mandaba primero el aviso de "vence hoy").
function etapaPendienteFila(f, hoy) {
  if (f.cobrado) return -1;
  var dias = diasDesdeIso(f.fecha_pago, hoy); if (dias === null) return -1;
  for (var i = ETAPAS_COBRO.length - 1; i >= 0; i--) {
    if (dias >= ETAPAS_COBRO[i].dias) return marcaHecha(f.marcas[i]) ? -1 : i;
  }
  return -1;
}

// Deja en la hoja el mismo texto que ve el admin (VENCIDO / ENVIADO) para poder leerla a mano.
function actualizarEstadosVencimientoSaaS(hoy) {
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);
  if (!filas.length) return;
  var primera = filas[0].fila; var ultima = filas[filas.length - 1].fila;
  var rango = hoja.getRange(primera, 7, ultima - primera + 1, 3); var vals = rango.getValues();
  for (var i = 0; i < filas.length; i++) {
    var est = estadosVencimientoFila(filas[i], hoy); var k = filas[i].fila - primera;
    vals[k] = est;
  }
  rango.setValues(vals);
  invalidarCacheControl();
}

// Lista de clientes para el selector: se muestra el nombre y se guarda el correo.
function listarCorreosUsuarios() {
  var mapa = mapaUsuarios(true); var lista = [];
  for (var c in mapa) { lista.push({ correo: mapa[c].correo, nombre: mapa[c].nombre }); }
  lista.sort(function(a, b) { return (a.nombre || a.correo).toLowerCase().localeCompare((b.nombre || b.correo).toLowerCase()); });
  return lista;
}

// correo (minúsculas) → { uid, nombre, correo }
function mapaUsuarios(forzar) {
  if (_mapaUsuarios && !forzar) return _mapaUsuarios;
  var db = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var data = db.getRange("A2:P" + Math.max(2, db.getLastRow())).getValues(); var mapa = {};
  for (var i = 0; i < data.length; i++) {
    var correo = String(data[i][15] || "").trim();
    // si dos cuentas declaran el mismo correo, gana la primera registrada (evita que alguien se "robe" la suscripción de otro)
    var k = correo.toLowerCase();
    if (data[i][0] && correo && !mapa[k]) mapa[k] = { uid: data[i][0], nombre: String(data[i][1] || ""), correo: correo };
  }
  _mapaUsuarios = mapa; return mapa;
}
function usuarioPorCorreo(correo) { return correo ? (mapaUsuarios()[String(correo).trim().toLowerCase()] || null) : null; }

function obtenerDatosControlSaaS(payload) {
  try {
    migrarControlSaaSV2();
    migrarControlSaaSV3();
    var hoy = hoyIso(payload && payload.hoy);
    var filas = leerFilasControl(getHojaControl());
    var salida = filas.map(function(f) {

}

// Al cambiar la fecha de suscripción o el tipo se recalcula la fecha de pago; al marcar Cobrado sin fecha de pago se usa hoy.
function completarCambiosControl(fila, cambios, hoy) {
  var r = {}; for (var k in cambios) { r[k] = cambios[k]; }
  var cambiaCiclo = ("fecha_suscripcion" in cambios) || ("tipo_suscripcion" in cambios);
  var inicio = ("fecha_suscripcion" in cambios) ? cambios.fecha_suscripcion : (fila ? fila.fecha_suscripcion : "");
  if (cambiaCiclo && inicio && !("fecha_pago" in cambios)) {
    var tipo = cambios.tipo_suscripcion || (fila ? fila.tipo_suscripcion : "") || "Mensual";
    r.fecha_pago = fechaPagoDeCiclo(inicio, tipo);
  }
  if (cambios.cobrado === true && !(fila && fila.fecha_pago) && !r.fecha_pago) r.fecha_pago = hoy;
  return r;
}

function escribirCambiosFila(hoja, nFila, cambios) {
  for (var campo in cambios) {
    var col = CAMPOS_CONTROL[campo]; if (!col) continue;
    var v = cambios[campo]; var celda = hoja.getRange(nFila, col);
    if (campo === "fecha_suscripcion" || campo === "fecha_pago") {
      if (v) { var f = parseFechaIso(v); if (!f) throw "Fecha inválida."; celda.setValue(f); } else celda.clearContent();
    }
    else if (campo === "cobrado") celda.setValue(v === true);
    else if (campo === "monto") celda.setValue(Math.max(0, round2(parseFloat(v) || 0)));
    else if (campo === "tipo_suscripcion") celda.setValue(v === "Anual" ? "Anual" : "Mensual");
    else celda.setValue(textoSeguro(v, campo === "comprobante_url" ? 500 : 120));
  }
  invalidarCacheControl();
}

function guardarCambioControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var previo = buscarFilaControl(hoja, String(p.id || ""));
  if (!previo) throw "No se encontró el registro. Recarga la pantalla.";
  var cambios = completarCambiosControl(previo, p.cambios || {}, hoy);
  escribirCambiosFila(hoja, previo.fila, cambios);
  // Como el trigger "reiniciar_avisos": al marcar cobrado o cambiar la fecha de pago se limpian los avisos enviados.
  if (cambios.cobrado === true || ("fecha_pago" in cambios && cambios.fecha_pago !== previo.fecha_pago)) hoja.getRange(previo.fila, 7, 1, 3).clearContent();
  invalidarCacheControl();
  var nueva = buscarFilaControl(hoja, previo.id);
  sincronizarControlConFinanzas(nueva, previo, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Guardado";
}

function insertarFilaControl(hoja, o) {

  hoja.getRange(fila, 1, 1, CONTROL_COLS).setValues([[textoSeguro(o.plan, 120), textoSeguro(o.correo, 120), Math.max(0, round2(Number(o.monto) || 0)), o.cobrado === true, o.fecha_suscripcion ? parseFechaIso(o.fecha_suscripcion) : "", o.fecha_pago ? parseFechaIso(o.fecha_pago) : "", "", "", "", textoSeguro(o.comprobante_url, 500), "", "", textoSeguro(o.metodo_pago, 60), o.tipo_suscripcion === "Anual" ? "Anual" : "Mensual", id]]);
  invalidarCacheControl();
  return id;
}

// La fila vacía del final de la tabla: al escribir en ella se crea el registro.
function crearRegistroControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var base = { plan: "", correo: "", monto: 0, cobrado: false, tipo_suscripcion: "Mensual", metodo_pago: "", fecha_suscripcion: "", fecha_pago: "", comprobante_url: "" };

  sincronizarControlConFinanzas(buscarFilaControl(hoja, id), null, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Registro creado";
}

// Renovar: el nuevo ciclo empieza donde terminó el anterior (fecha de pago en mensual, un año después en anual).
// Renovar: el nuevo ciclo empieza donde terminó el anterior (fecha de pago en mensual, un año después en anual).
// Nota de diseño: el mensual se cobra al FINAL del ciclo (fecha de pago = inicio + 1 mes) y el anual al INICIO (fecha de pago = inicio).
function renovarControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var f = buscarFilaControl(hoja, String(p.id || ""));
  if (!f) throw "No se encontró el registro. Recarga la pantalla.";
  var ciclo = siguienteCiclo(f.fecha_suscripcion, f.fecha_pago, f.tipo_suscripcion, hoy);
  var nuevoMonto = f.monto;
  if (p.nuevoMonto !== null && p.nuevoMonto !== undefined && p.nuevoMonto !== "") {
    nuevoMonto = Number(p.nuevoMonto);
    if (isNaN(nuevoMonto) || nuevoMonto < 0) throw "Monto inválido.";
  }
  // Evita renovar dos veces el mismo periodo (doble clic)
  var existe = leerFilasControl(hoja).some(function(x) {
    return x.id !== f.id && x.correo.toLowerCase() === f.correo.toLowerCase() && x.plan === f.plan && x.fecha_suscripcion === ciclo.fecha_suscripcion;
  });
  if (existe) throw "Ya existe un renglón para ese periodo. Edítalo en lugar de renovar de nuevo.";
  var id = insertarFilaControl(hoja, { plan: f.plan, correo: f.correo, monto: nuevoMonto, cobrado: false, metodo_pago: f.metodo_pago, tipo_suscripcion: f.tipo_suscripcion, fecha_suscripcion: ciclo.fecha_suscripcion, fecha_pago: ciclo.fecha_pago });
  sincronizarControlConFinanzas(buscarFilaControl(hoja, id), null, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Se ha generado el nuevo periodo.";
}

function eliminarRegistroControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var f = buscarFilaControl(hoja, String(p.id || ""));
  if (!f) throw "No se encontró el registro. Recarga la pantalla.";
  hoja.deleteRow(f.fila);
  invalidarCacheControl();
  sincronizarControlConFinanzas(null, f, hoy);
  return "Registro eliminado";
}

  else { var enc = h.getRange(1, 10, 1, 2).getValues()[0]; if (!enc[0]) h.getRange(1, 10).setValue("CONTROL_ID"); if (!enc[1]) h.getRange(1, 11).setValue("CORREO"); }

function filaPagoPorControl(hojaPagos, controlId) {
  var n = hojaPagos.getLastRow(); if (n < 2 || !controlId) return 0;
  var ids = hojaPagos.getRange(2, 10, n - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === controlId) return i + 2; }
  return 0;
}
function insertarPagoDeControl(hojaPagos, f, usuario) {
  var fecha = f.fecha_pago || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var d = parseFechaIso(fecha);
  hojaPagos.appendRow([usuario.uid, usuario.nombre, d, MESES_UP[d.getMonth()], d.getFullYear(), f.tipo_suscripcion, f.metodo_pago || "", f.comprobante_url || "", f.monto, f.id, usuario.correo]);
}

function recalcularSuscripcionSaaS(correo) {
  var usuario = usuarioPorCorreo(correo); if (!usuario) return;
  CacheService.getScriptCache().remove("dash_" + usuario.uid);   // el cliente ve su estado de pago al instante
  var hojaEstado = getHojaEstado(); var uid = usuario.uid;
  var filas = leerFilasControl(getHojaControl()).filter(function(f) { return f.correo.toLowerCase() === usuario.correo.toLowerCase(); });
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var tienePagos = false;
  if (np >= 2) { var uids = hojaPagos.getRange(2, 1, np - 1, 1).getValues(); for (var i = 0; i < uids.length; i++) { if (uids[i][0] === uid) { tienePagos = true; break; } } }

  var ne = hojaEstado.getLastRow(); var filaEstado = 0;
  if (ne >= 2) { var ue = hojaEstado.getRange(2, 1, ne - 1, 1).getValues(); for (var j = 0; j < ue.length; j++) { if (ue[j][0] === uid) { filaEstado = j + 2; break; } } }

  if (!filas.length && !tienePagos) { if (filaEstado) hojaEstado.deleteRow(filaEstado); return; }
  var pendiente = null, ultimo = null;
  filas.forEach(function(f) {
    if (!f.fecha_pago) return;
    if (!f.cobrado && (!pendiente || f.fecha_pago < pendiente.fecha_pago)) pendiente = f;
    if (f.cobrado && (!ultimo || f.fecha_pago > ultimo.fecha_pago)) ultimo = f;
  });
  var plan, proximo, base;
  if (pendiente) { plan = pendiente.tipo_suscripcion; proximo = pendiente.fecha_pago; base = pendiente.fecha_suscripcion; }
  else if (ultimo) { plan = ultimo.tipo_suscripcion; proximo = sumarMesesIso(ultimo.fecha_pago, plan === "Anual" ? 12 : 1); base = ultimo.fecha_suscripcion || ultimo.fecha_pago; }
  else return; // solo hay pagos antiguos sin filas de control: se deja como está
  var valores = [[uid, base ? parseFechaIso(base) : "", plan, parseFechaIso(proximo)]];
  if (filaEstado) hojaEstado.getRange(filaEstado, 1, 1, 4).setValues(valores);
  else hojaEstado.getRange(Math.max(2, ne + 1), 1, 1, 4).setValues(valores);
}

function actualizarPagoDeControl(hojaPagos, fila, f) {
  var d = parseFechaIso(f.fecha_pago) || parseFechaIso(hoyIso());
  var actual = hojaPagos.getRange(fila, 7, 1, 3).getValues()[0];   // G método · H recibo · I monto
  // C fecha · D mes · E año · F plan · G método · H recibo · I monto
  hojaPagos.getRange(fila, 3, 1, 7).setValues([[d, MESES_UP[d.getMonth()], d.getFullYear(), f.tipo_suscripcion, f.metodo_pago || actual[0], f.comprobante_url || actual[1], f.monto]]);
}

function sincronizarControlConFinanzas(nueva, previa, hoy) {
  var hojaPagos = getHojaPagos();
  if (!nueva) { // fila eliminada: el pago se conserva en el historial, solo se desvincula
    var fp = filaPagoPorControl(hojaPagos, previa.id); if (fp) hojaPagos.getRange(fp, 10).setValue("");
    recalcularSuscripcionSaaS(previa.correo); return;
  }
  // Si el correo todavía no está registrado el pago se guarda igual (sin UID) y se enlaza cuando el usuario se registre
  var usuario = usuarioPorCorreo(nueva.correo) || { uid: "", nombre: "", correo: nueva.correo };
  var registrado = !!usuarioPorCorreo(nueva.correo);
  if (previa && previa.correo.toLowerCase() !== nueva.correo.toLowerCase()) {
    var fm = filaPagoPorControl(hojaPagos, nueva.id);
    if (fm) { hojaPagos.getRange(fm, 1, 1, 2).setValues([[usuario.uid, usuario.nombre]]); hojaPagos.getRange(fm, 11).setValue(usuario.correo); }
    if (previa.correo) recalcularSuscripcionSaaS(previa.correo);
  }
  if (nueva.cobrado) {
    var fila = filaPagoPorControl(hojaPagos, nueva.id);
    if (!fila) insertarPagoDeControl(hojaPagos, nueva, usuario);
    else if (previa && (previa.monto !== nueva.monto || previa.comprobante_url !== nueva.comprobante_url || previa.metodo_pago !== nueva.metodo_pago || previa.fecha_pago !== nueva.fecha_pago || previa.tipo_suscripcion !== nueva.tipo_suscripcion)) {
      actualizarPagoDeControl(hojaPagos, fila, nueva);   // ahora también se sincronizan fecha, mes, año y tipo de plan
    }
  } else if (previa && previa.cobrado && !nueva.cobrado) {
    var fe = filaPagoPorControl(hojaPagos, nueva.id); if (fe) hojaPagos.deleteRow(fe);
  }
  if (registrado) recalcularSuscripcionSaaS(nueva.correo);
}

// Una sola vez: IDs, tipo "Mensual" por defecto y pago en el historial de cada cuota ya cobrada.
function migrarControlSaaSV2() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty("CONTROL_SAAS_V2") === "ok") return;
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);
  if (filas.length) {
    var primera = filas[0].fila; var ultima = filas[filas.length - 1].fila;
    var rango = hoja.getRange(primera, 14, ultima - primera + 1, 1); var vals = rango.getValues();
    filas.forEach(function(f) { if (vals[f.fila - primera][0] === "") vals[f.fila - primera][0] = "Mensual"; });
    rango.setValues(vals);
  }
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var pagos = np >= 2 ? hojaPagos.getRange(2, 1, np - 1, 11).getValues() : [];
  filas.forEach(function(f) {
    var usuario = usuarioPorCorreo(f.correo); if (!f.cobrado || !usuario) return;
    if (filaPagoPorControl(hojaPagos, f.id)) return;
    var d = parseFechaIso(f.fecha_pago); var vinculado = 0;
    for (var i = 0; i < pagos.length; i++) { // pago manual antiguo del mismo usuario, mes y monto: se vincula en vez de duplicarlo
      if (pagos[i][0] === usuario.uid && !pagos[i][9] && d && String(pagos[i][3]).toUpperCase() === MESES_UP[d.getMonth()] && Number(pagos[i][4]) === d.getFullYear() && Number(pagos[i][8]) === f.monto) { vinculado = i + 2; pagos[i][9] = f.id; break; }
    }
    if (vinculado) hojaPagos.getRange(vinculado, 10, 1, 2).setValues([[f.id, usuario.correo]]);
    else insertarPagoDeControl(hojaPagos, f, usuario);
  });
  var vistos = {};
  filas.forEach(function(f) { var k = f.correo.toLowerCase(); if (f.correo && !vistos[k]) { vistos[k] = true; recalcularSuscripcionSaaS(f.correo); } });
  invalidarCacheControl();
  props.setProperty("CONTROL_SAAS_V2", "ok");
}

// Una sola vez: cuotas cobradas sin pago en el historial (correo no registrado en su momento).
function migrarControlSaaSV3() {
  var props = PropertiesService.getScriptProperties();

  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);

    if (!f.cobrado || !f.id || existentes[f.id]) return;

    insertarPagoDeControl(hojaPagos, f, usuario);
    existentes[f.id] = true;
  });
  props.setProperty("CONTROL_SAAS_V3", "ok");
}

// Asunto sobrio (sin emojis) + texto y HTML + pie que identifica quién envía: toleran mejor los filtros anti-spam.
function construirCorreoCobro(d) {
  var concepto = String(d.concepto || "Pendiente").replace(/[\r\n]+/g, " ").trim().substring(0, 80);
  var firma = d.remitente || d.responderA;
  var pie = "Recibes este mensaje de " + firma + " por un pago pendiente. Si ya realizaste el pago, ignora este recordatorio o responde a este correo.";
  var cuerpo = String(d.cuerpo).trim();
  var text = cuerpo + "\n\nSaludos,\n" + firma + "\n\n--\n" + pie;
  var parrafos = cuerpo.split(/\n{2,}/).map(function(p) { return '<p style="margin:0 0 14px">' + escaparHtml(p).replace(/\n/g, "<br>") + "</p>"; }).join("");
  var html = '<div style="font-family:Arial,Helvetica,sans-serif;font-size:15px;line-height:1.5;color:#222;max-width:560px">' + parrafos +
    '<p style="margin:18px 0 0">Saludos,<br>' + escaparHtml(firma) + "</p>" +
    '<hr style="border:none;border-top:1px solid #ddd;margin:22px 0 10px">' +
    '<p style="margin:0;font-size:12px;color:#777">' + escaparHtml(pie) + "</p></div>";
  return { subject: "Recordatorio de pago de tu plan: " + concepto, text: text, html: html };
}

function ejecutarEnvioCobroSaaS(p) {
  var hoy = hoyIso(p && p.hoy);
  var hojaScripts = obtenerHojaScripts();
  var scripts = [hojaScripts.getRange("B6").getValue(), hojaScripts.getRange("B14").getValue(), hojaScripts.getRange("B22").getValue()];
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja); var pendientes = [];
  filas.forEach(function(f) { var et = etapaPendienteFila(f, hoy); if (et >= 0) pendientes.push({ f: f, etapa: et }); });
  if (!pendientes.length) { actualizarEstadosVencimientoSaaS(hoy); return { mensaje: "No hay recordatorios pendientes por enviar.", enviados: 0 }; }

  var rem = obtenerConfigRemitente(); var mapa = mapaUsuarios();
  var cuota = MailApp.getRemainingDailyQuota();
  var enviados = 0, sinCorreo = 0, sinPlantilla = 0, sinCuota = 0, restantes = 0; var fallidos = []; var primero = true;

  for (var i = 0; i < pendientes.length; i++) {
    var f = pendientes[i].f; var etapa = pendientes[i].etapa;
    if (!f.correo || f.correo.indexOf("@") < 0) { sinCorreo++; continue; }
    if (!scripts[etapa]) { sinPlantilla++; continue; }
    if (enviados >= cuota) { sinCuota++; continue; }
    if (enviados >= MAX_ENVIOS_POR_EJECUCION) { restantes++; continue; }   // lote máximo por ejecución: vuelve a ejecutar para el resto
    var u = mapa[f.correo.toLowerCase()];
    var nombre = (u && u.nombre) ? u.nombre : f.correo.split("@")[0];
    var texto = rellenarPlantillaCobro(scripts[etapa], { nombre: nombre, proyecto: f.plan || "Suscripción", monto: f.monto });
    var correo = construirCorreoCobro({ concepto: f.plan || "Suscripción", cuerpo: texto, remitente: rem.nombre, responderA: rem.direccion });
    if (!primero) Utilities.sleep(800); // una ráfaga idéntica se parece a spam masivo
    primero = false;
    try {
      var op = { htmlBody: correo.html, replyTo: rem.direccion };
      if (rem.nombre) op.name = rem.nombre;
      if (rem.alias) op.from = rem.alias;
      GmailApp.sendEmail(f.correo, correo.subject, correo.text, op);
      enviados++;
      hoja.getRange(f.fila, ETAPAS_COBRO[etapa].col).setValue("ENVIADO");
      // las etapas anteriores que nunca se enviaron quedan como OMITIDO (no se mandará un aviso viejo después)
      for (var k = 0; k < etapa; k++) { if (!marcaHecha(f.marcas[k])) hoja.getRange(f.fila, ETAPAS_COBRO[k].col).setValue("OMITIDO"); }
    } catch (e) { fallidos.push(f.correo + " (" + String(e.message || e).split("\n")[0].substring(0, 200) + ")"); }
  }
  invalidarCacheControl();
  actualizarEstadosVencimientoSaaS(hoy);
  var mensaje = "Se enviaron " + enviados + " recordatorios de cobro.";
  if (sinCorreo) mensaje += " " + sinCorreo + " vencidos no tienen un correo válido.";
  if (sinPlantilla) mensaje += " " + sinPlantilla + " no se enviaron porque su plantilla está vacía.";
  if (sinCuota) mensaje += " " + sinCuota + " quedaron pendientes por el límite diario de correos de Google.";
  if (restantes) mensaje += " Quedan " + restantes + " por enviar: vuelve a ejecutar el cobro.";
  if (fallidos.length) mensaje += " Fallaron: " + fallidos.join("; ") + ".";
  return { mensaje: mensaje, enviados: enviados };
}

// Los reportes leen FINANZAS_PAGOS (la misma fuente del panel de Finanzas), así los totales siempre coinciden.
function leerPagosParaReporte() {
  migrarControlSaaSV2(); migrarControlSaaSV3();
  var hp = getHojaPagos(); var n = hp.getLastRow(); if (n < 2) return [];
  var vals = hp.getRange(2, 1, n - 1, 11).getValues(); var out = [];
  vals.forEach(function(f) {
    if (!f[0] && !f[10]) return;
    var iso = aIsoFecha(f[2]); var anio, mes;
    if (iso) { anio = Number(iso.substring(0, 4)); mes = Number(iso.substring(5, 7)) - 1; }
    else { anio = Number(f[4]); mes = MESES_UP.indexOf(String(f[3]).toUpperCase()); }
    if (!anio || mes < 0) return;
    out.push({ anio: anio, mes: mes, monto: parseFloat(f[8]) || 0, cliente: String(f[10] || f[1] || "").trim() });
  });
  return out;
}

function obtenerDatosReporteAnualSaaS(anio) {
  try {
    var meses = []; for (var m = 0; m < 12; m++) meses.push(0); var total = 0;
    leerPagosParaReporte().forEach(function(pg) {
      if (pg.anio === Number(anio)) { meses[pg.mes] += pg.monto; total += pg.monto; }
    });
    return { meses: meses.map(round2), total: round2(total) };
  } catch (e) { return { error: mensajeError(e) }; }
}

function obtenerDatosReporteClientesSaaS(anio, mesTexto) {
  try {
    var mesIndex = MESES_UP.indexOf(String(mesTexto).toUpperCase()); var mapa = {}; var orden = [];
    leerPagosParaReporte().forEach(function(pg) {
      if (pg.anio === Number(anio) && pg.mes === mesIndex) {
        var cliente = pg.cliente || "Sin Usuario";
        if (mapa[cliente] === undefined) { mapa[cliente] = 0; orden.push(cliente); }
        mapa[cliente] += pg.monto;
      }
    });
    return { labels: orden, valores: orden.map(function(c) { return round2(mapa[c]); }) };
  } catch (e) { return { error: mensajeError(e) }; }
}
