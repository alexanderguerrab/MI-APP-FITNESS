function isUserAdmin(userId) {
  if (userId === ADMIN_UID) return true;
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var data = dbSheet.getRange("A2:O" + Math.max(2, dbSheet.getLastRow())).getValues();
  for(var i=0; i<data.length; i++) {

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000); 
    
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var userId = data.userId;
    var payload = data.datos || {};

    if (!userId) return ContentService.createTextOutput(JSON.stringify({ error: "Falta userId" })).setMimeType(ContentService.MimeType.JSON);

    var resultado = "";
    switch (action) {
      case "getDashboardData": resultado = getDashboardData(userId, payload.email); break;
      case "saveWelcomeData": resultado = saveWelcomeData(payload, userId); break;
      case "guardarEscalon": resultado = guardarEscalon(payload, userId); break;
      case "guardarFotoEscalonBackend": resultado = guardarFotoEscalonBackend(payload, userId); break;
      case "saveMedidasMesUniversal": resultado = saveMedidasMesUniversal(payload, userId); break;
      case "saveDiaDiario": resultado = saveDiaDiario(payload, userId); break;
      case "registrarTelefono": resultado = registrarTelefono(payload, userId); break;
      
      // SOLO SUPER ADMIN
      case "guardarFotosCalculadora": 
        if (userId !== ADMIN_UID) throw "No autorizado. Solo Super Admin.";
        resultado = guardarFotosCalculadora(payload); break;
      case "saveRecetaBackend": 
        if (userId !== ADMIN_UID) throw "Solo administrador. Solo Super Admin.";
        resultado = saveRecetaBackend(payload); break;
      case "actualizarIconoApp": 
        if (userId !== ADMIN_UID) throw "No autorizado";
        resultado = actualizarIconoApp(payload.url); break;
        
      // ADMIN Y SUPER ADMIN
      case "getUsersList": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = getUsersList(); break;
      case "cambiarEstadoUser": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = cambiarEstadoUser(payload.uid, payload.estado); break;
      case "eliminarUser": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = eliminarUser(payload.uid); break;
      case "getFinanzasData": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = getFinanzasData(); break;
      case "registrarPago": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = registrarPago(payload); break;
      case "eliminarPago": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = eliminarPago(payload); break;
      case "editarPagoBackend": 
        if (!isUserAdmin(userId)) throw "No autorizado";
        resultado = editarPagoBackend(payload); break;
      default: resultado = { error: "Acción no válida" };
    }

    return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    var errMsg = error.toString().includes("Lock timeout") ? "Servidor saturado. Intenta de nuevo en unos segundos." : error.toString();
    return ContentService.createTextOutput(JSON.stringify({ error: errMsg })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function registrarTelefono(payload, userId) {
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  dbSheet.getRange("R1").setValue("TELEFONO"); 
  var uids = dbSheet.getRange("A2:A").getValues();
  var targetRow = -1;
  for (var i = 0; i < uids.length; i++) { 
    if (uids[i][0] == userId) { targetRow = i + 2; break; } 
  }
  
  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    var estadoInicial = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    dbSheet.getRange(targetRow, 15).setValue(estadoInicial); 
    if (payload.email) dbSheet.getRange(targetRow, 16).setValue(payload.email); 
  }
  
  dbSheet.getRange(targetRow, 18).setValue(payload.telefono); 
  return "Teléfono guardado";
}

function getUserSpreadsheet(userId, email, nombre) {
  var centralSs = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = centralSs.getSheetByName("DB_Perfiles");
  var data = dbSheet.getRange("A2:Q").getValues(); 
  var targetRow = -1;
  var sheetId = "";
  var estado = "Autorizado";

  for (var i = 0; i < data.length; i++) {
    if (data[i][0] == userId) {
      targetRow = i + 2;
      estado = data[i][14] || "Autorizado"; 
      sheetId = data[i][16]; 
      
      if (email && data[i][15] === "") dbSheet.getRange(targetRow, 16).setValue(email);
      break;
    }
  }

  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    estado = (userId === ADMIN_UID) ? "Autorizado" : "Pendiente";
    dbSheet.getRange(targetRow, 1).setValue(userId); 
    dbSheet.getRange(targetRow, 15).setValue(estado); 
    if (email) dbSheet.getRange(targetRow, 16).setValue(email); 
  }

  if (!sheetId || sheetId === "") {
    var template = DriveApp.getFileById(TEMPLATE_ID);
    var targetFolder = DriveApp.getFolderById(FOLDER_ID);
    var copy = template.makeCopy("Fitness Pro - " + (email || userId), targetFolder);
    sheetId = copy.getId();
    dbSheet.getRange(targetRow, 17).setValue(sheetId);
  }

  return { ss: SpreadsheetApp.openById(sheetId), estado: estado, esNuevo: (sheetId === "") };
}

function actualizarDBPerfiles(userId, pSheet, nombre) {
  var dbSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DB_Perfiles");
  var uids = dbSheet.getRange("A2:A").getValues();
  var targetRow = -1;
  for (var i = 0; i < uids.length; i++) { if (uids[i][0] == userId) { targetRow = i + 2; break; } }
  if (targetRow === -1) return;

  var pesoIni = pSheet.getRange("A10").getValue();
  var talla = parseFloatSafe(pSheet.getRange("A7").getValue());
  var pesoId = pSheet.getRange("A13").getValue();
  var fNac = pSheet.getRange("A19").getDisplayValue();
  var pObj = pSheet.getRange("A16").getValue();
  var avatarUrl = readImageUrl(pSheet, "A21");
  var pesoAct = parseFloatSafe(pSheet.getRange("K31").getValue());
  var evolucion = pSheet.getRange("K27").getValue();
  
  var edadVal = pSheet.getRange("A4").getValue();
  var edad = parseFloatSafe(edadVal);
  var today = new Date(); today.setHours(0,0,0,0);
  
  if (!edad && fNac) {
     var fNacDate = fNac;
     if (typeof fNac === "string" && fNac.indexOf("/") !== -1) {
         var parts = fNac.split("/");
         if (parts.length === 3) fNacDate = new Date(parts[2], parts[1]-1, parts[0]);
     }
     if (fNacDate && !isNaN(new Date(fNacDate).getTime())) {
         edad = Math.floor((today - new Date(fNacDate)) / (365.25 * 24 * 60 * 60 * 1000));
     }
  }
  
  var calcSheet = pSheet.getParent().getSheetByName("Calculadora") || pSheet.getParent().getSheetByName("Calculadoras");
  var genero = calcSheet ? calcSheet.getRange("D2").getValue() : "Hombre";
  
  var deficit = pSheet.getRange("C19").getValue(); 
  if (pesoAct && talla && edad) {
      var tmb = (10 * pesoAct) + (6.25 * (talla * 100)) - (5 * edad) + (genero === "Mujer" ? -161 : 5);
      deficit = Math.round((tmb * 1.2) - 500);
      
      var diarioSheet = pSheet.getParent().getSheetByName("Diario");
      if(diarioSheet) diarioSheet.getRange("K382").setValue(deficit);
  }
  
  var rowData = [[nombre || pSheet.getRange("A2").getValue(), pesoIni, talla, pesoId, fNac, deficit, pObj, avatarUrl, pesoAct, evolucion]];
  dbSheet.getRange(targetRow, 2, 1, 10).setValues(rowData);
}

function getFinanzasData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_Perfiles");
  var usersData = dbSheet.getRange("A2:B" + Math.max(2, dbSheet.getLastRow())).getValues();
  
  var estadoSheet = ss.getSheetByName("FINANZAS_ESTADO");
  var estadoData = estadoSheet ? estadoSheet.getRange("A2:D" + Math.max(2, estadoSheet.getLastRow())).getValues() : [];
  var estadosMap = {};
  for(var i = 0; i < estadoData.length; i++) { if(estadoData[i][0]) estadosMap[estadoData[i][0]] = { base: estadoData[i][1], plan: estadoData[i][2], proximo: estadoData[i][3] }; }

  var pagosSheet = ss.getSheetByName("FINANZAS_PAGOS");
  var pagosData = pagosSheet ? pagosSheet.getRange("A2:I" + Math.max(2, pagosSheet.getLastRow())).getDisplayValues() : [];

  var usuarios = [];
  var alertas = [];
  var hoy = new Date(); hoy.setHours(0,0,0,0);
  var limiteAlerta = new Date(hoy); limiteAlerta.setDate(limiteAlerta.getDate() + 5);

  for(var j = 0; j < usersData.length; j++) {
    var uid = usersData[j][0];
    if(!uid) continue;
    var nombre = usersData[j][1] || "Sin Nombre";
    var est = estadosMap[uid] || { base: "", plan: "Mensual", proximo: "" };
    
    usuarios.push({ uid: uid, nombre: nombre, base: est.base ? Utilities.formatDate(new Date(est.base), "GMT-4", "yyyy-MM-dd") : "", plan: est.plan, proximo: est.proximo ? Utilities.formatDate(new Date(est.proximo), "GMT-4", "dd/MM/yyyy") : "Sin pagos" });

    if(est.proximo && est.proximo instanceof Date) {
       var fProx = new Date(est.proximo);
       if(fProx < hoy) alertas.push({ nombre: nombre, fecha: Utilities.formatDate(fProx, "GMT-4", "dd/MM/yyyy"), estado: "VENCIDO" });
       else if(fProx <= limiteAlerta) alertas.push({ nombre: nombre, fecha: Utilities.formatDate(fProx, "GMT-4", "dd/MM/yyyy"), estado: "PRÓXIMO A VENCER" });
    }
  }

  var pagosConFila = [];
  for(var p = 0; p < pagosData.length; p++) {
    if(pagosData[p][0]) {
      pagosConFila.push({
        fila: p + 2,
        uid: pagosData[p][0],
        nombre: pagosData[p][1],
        fechaPago: pagosData[p][2],
        mes: pagosData[p][3],
        anio: pagosData[p][4],
        plan: pagosData[p][5],
        metodo: pagosData[p][6],
        recibo: pagosData[p][7],
        costo: pagosData[p][8] || ""
      });
    }
  }
  pagosConFila.reverse();

  usuarios.sort(function(a, b) { return a.nombre.localeCompare(b.nombre); });
  return { usuarios: usuarios, alertas: alertas, pagos: pagosConFila };
}

  var fBaseParts = payload.fechaBase.split('-');
  var fBase = new Date(fBaseParts[0], fBaseParts[1] - 1, fBaseParts[2]);
  
  var mesesArray = ["ENERO","FEBRERO","MARZO","ABRIL","MAYO","JUNIO","JULIO","AGOSTO","SEPTIEMBRE","OCTUBRE","NOVIEMBRE","DICIEMBRE"];
  var mesIndex = mesesArray.indexOf(payload.mes); 
  
  var fProximo;
  if (payload.plan === "Anual") {
     fProximo = new Date(parseInt(payload.anio) + 1, mesIndex, fBase.getDate());
  } else {
     fProximo = new Date(parseInt(payload.anio), mesIndex + 1, fBase.getDate()); 
  }

  if(fila === -1) {
     fila = estadoSheet.getLastRow() + 1;
     estadoSheet.getRange(fila, 1, 1, 4).setValues([[payload.uid, fBase, payload.plan, fProximo]]);
  } else {
     estadoSheet.getRange(fila, 2, 1, 3).setValues([[fBase, payload.plan, fProximo]]);
  }
  return "Pago sincronizado con éxito";
}

function eliminarPago(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var pagosSheet = ss.getSheetByName("FINANZAS_PAGOS");
  pagosSheet.deleteRow(payload.fila);
  return "Pago eliminado exitosamente de la base de datos";
}

  var fParts = payload.fechaPago.split('-');
  var fObj = fParts.length === 3 ? new Date(fParts[0], fParts[1] - 1, fParts[2]) : new Date();
